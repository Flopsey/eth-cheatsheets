\subsection{Grundbegriffe der W'theorie}

% \emph{Zufallsexperimente} sind Experimente, deren Ergebnisse nicht immer exakt vorausgesagt werden können. Dafür möchten wir ein mathematisches Modell.

% \begin{mainbox}{Grundraum}
%     Der \emph{Ereignisraum} oder \emph{Grundram \( \Omega \neq 0 \)} ist die Menge aller möglichen Ergebnisse des betrachteten Zufallsexperiments. Die Elemente \( \omega \in \Omega \) heissen \emph{Elementarereignisse} oder \emph{Ausgänge des Experiments}.
% \end{mainbox}

% \begin{mainbox}{Potenzmenge und Ereignisse}
%     % Die \emph{Potenzmenge} von \( \Omega \), \( \mathcal{P}(\Omega) \) oder \( 2^\Omega \), ist die Menge aller Teilmengen von \( \Omega \).

%     Ein \emph{prinzipielles Ereignis} ist eine Teilmenge \( \mathcal{A} \subseteq \Omega \), also eine Kollektion von Elementarereignissen.

%     Die Klasse aller \emph{(beobachtbaren) Ereignisse} bezeichnen wir mit \( \F \). Das ist eine Teilmenge der Potenzmenge von \( \Omega \).
% \end{mainbox}

Ist \( \Omega \) endlich oder abzählbar, wählt man oft \( \F = \mathcal{P}(\Omega) \).
Ist \( \Omega \) überabzählbar, so muss \( \F \) eine echte Teilklasse von \( \mathcal{P}(\Omega) \) sein.
% In jedem Fall muss \( \F \) gewisse Axiome erfüllen:
\begin{mainbox}{\( \sigma \)-Algebra}
    \( \F \in \mathcal{P}(\Omega) \) nennt man eine \( \sigma \)-Algebra, wenn
    \begin{enumerate}
        \item \( \Omega \in \F \),
        \item für jedes \( A \in \F \) auch \( A^\complement \in \F \) ist,
        \item für jede \( (A_n)_{n \in \N} \) mit \( A_n \in \F, \ n \in \N\) auch \( \bigcup_{n \in \N} A_n \in \F \).
    \end{enumerate}
\end{mainbox}

Im Allgemeinen tritt jeweils \emph{genau ein} Ausgang (Elementarereignis \( \omega \)) ein.
\( A = \emptyset \) tritt niemals ein, \( A = \Omega \) tritt immer ein.

\begin{mainbox}{W'mass}
    Sei \( \Omega \) ein Grundraum und sei \( \F \) eine \( \sigma \)-Algebra. Eine Abbildung
    \[
        \P: \: \F \to [0, 1], \quad \text{mit } A \mapsto \P[A]
    \]
    heisst \emph{W'mass} auf \( (\Omega, \F) \), wenn die folgenden Axiome erfüllt sind:
    
    \begin{enumerate}
        \item \emph{Normiertheit}: \( \P[\Omega] = 1 \),
        \item \emph{\( \sigma \)-Additivität}: \( \P \left[ \bigcup_{n \in \N} A_n \right] = \sum_{n = 1}^\infty \P [A_n] \) für paarweise disjunkte Mengen \( A_n \), d.h. \( A_n \cap A_m = \emptyset \) für alle \( n \neq m \).
    \end{enumerate}

    % Zusammenfassend: Ein W'mass ordnet jedem Ereignis der \( \sigma \)-Algebra eine Zahl zwischen \( 0 \) und \( 1 \) (Wahrscheinlichkeit) zu.
\end{mainbox}

Aus diesen Axiomen lassen sich weitere Rechenregeln herleiten.
\begin{subbox}{}
    Für \( \P \) auf \( (\Omega, \F) \) und \( A, B \in \F \) gilt:
    \begin{itemize}
        \item \( \P [A^\complement] = 1 - \P[A] \), und insbesondere \( \P[\emptyset] = 0 \),
        \item \emph{Monotonie}: wenn \( A \subseteq B \), dann \( \P[A] \leq \P[B] \),
        \item \emph{Additionsregel}: \( \P[A] + \P[B] = \P[A \cup B] + \P[A \cap B] \)
    \end{itemize}
\end{subbox}

% \begin{mainbox}{W'raum}
%     Sei \( \Omega \) ein Grundraum, \( \F \) eine \(\sigma \)-Algebra und \( \P \) ein W'mass auf \( (\Omega, \F) \). Das Tripel \( (\Omega, \F, \P) \) heisst \emph{W'raum}.
% \end{mainbox}

\begin{subbox}{Messbarer Raum}
    % Allgemein verwenden wir in der Masstheorie folgende Terminolgie:
    \begin{itemize}
        \item Für eine \( \sigma \)-Algebra \( \mathcal{A} \) auf einer Grundmenge \( \Omega \) wird das Paar \( (\Omega, \mathcal{A}) \) ein \emph{messbarer Raum} genannt.
        \item Elemente \( A \in \mathcal{A} \), also Teilmengen \( A \subset \Omega \), die wir messen wollen, heissen \emph{messbare Mengen}.
        \item Auf messbaren Räumen (bzw. den \( \sigma \)-Algebren) lassen sich Masse \( \mu \) definieren. Diese sind im Allgemeinen nicht auf \( 1 \) normiert. Man verlangt stattdessen \( \mu(\emptyset) = 0 \).
        \item Das Tripel \( (\Omega, \mathcal{A}, \mu) \) heisst \emph{Massraum}.
        \item Ist das Mass normiert, \( \mu(\Omega) = 1 \), dann ist das Mass ein W'mass und der Massraum wird zum W'raum.
    \end{itemize}
\end{subbox}

\subsection{Diskrete W'räume}

\( \Omega = \{ \omega_1, \omega_2, \dots, \omega_N \} \) endlich oder \( \Omega = \{ \omega_1, \omega_2, \dots \} \) abzählbar:
\begin{itemize}
    \item Dann wählt man \( \F = \mathcal{P}(\Omega) \)
    \item Dann ist \emph{\( \P \) definiert durch \( p_n \coloneqq \P [\omega_n] \), \( n = 1, \dots, N \) bzw. \( n \in \N \)}. Denn für jede beliebige Menge \( A \subseteq \Omega \) gilt \( A \in \F \) und
    \[
        \P[A] = \P \left[ \bigcup_{\omega_n \in A} \{ \omega_n \} \right] = \sum_{\omega_n \in A} \P [\{\omega_n\}] = \sum_{\{n | \omega_n \in A\}} p_n
    \]
\end{itemize}

\begin{mainbox}{Laplace-Modell}
    Sei \( \Omega = \{ \omega_1, \dots, \omega_N\} \) ein endlicher Grundraum. \( (\Omega, \F, \P) \) heisst \emph{Laplace-Modell} auf \( \Omega \), wenn
    \begin{itemize}
        \item \( \F = \mathcal{P}(\Omega) \),
        \item \( \P \) ist die \emph{diskrete Gleichverteilung} auf \( \Omega \). Insbesondere gilt für beliebige \( A \subseteq \Omega \)
        \[
            \P[A] = \frac{|A|}{|\Omega|} = \frac{\text{Anzahl Elementarereignisse in \(A\)}}{\text{Anzahl Elementarereignisse in \(\Omega\)}}
        \]
    \end{itemize}
\end{mainbox}
% Zur Berechnung von \( \P[A] \) muss man also die Elemente in \( A \) und in \( \Omega \) zählen. Das ist nicht immer leicht und benötigt oft \emph{Kombinatorik}.

\paragraph{Achtung!} Ein W'mass durch die Wahrscheinlichkeiten \( p_\omega \) jedes Elementarereignisses \( \omega \) zu definieren, funktioniert nur wenn \( \Omega \) endlich oder abzählbar ist.
Es \emph{scheitert}, wenn \( \Omega \) \emph{nicht abzählbar} ist.
Deshalb verwenden wir die axiomatische Definition des W'masses und als natürliche Wahl des W'raums \( \Omega = [0, 1] \), \( \F =  \text{Borel \( \sigma \)-Algebra} \), \( \P[A] = \) Lebesguemass von \( A \).

\subsection{Bedingte Wahrscheinlichkeiten}

\begin{mainbox}{Bedingte Wahrscheinlichkeit}
    Sei \( (\Omega, \F, \P) \) ein W'raum. Seien \( A, \, B \) zwei Ereignisse mit \( \P[B] > 0 \). Die \emph{bedingte Wahrscheinlichkeit von \( A \) gegeben \( B \)} ist:
    \[
        \P[A \: | \: B] \coloneqq \frac{\P[A \cap B]}{\P[B]}
    \]
\end{mainbox}

\begin{mainbox}{}
    Sei \( (\Omega, \F, \P) \) ein W'raum. Sei \( B \) ein Ereignis mit positiver Wahrscheinlichkeit. Dann ist \( \P^*: \: \F \to [0, 1] \) definiert durch
    \[
        A \mapsto \P^*[A] \coloneqq \P[A \: | \: B]
    \]
    wieder ein W'mass auf \( (\Omega, \F) \) \text{.}
\end{mainbox}

Ist \( \Omega \) endlich oder abzählbar mit \( \F = \mathcal{P} \), dann ist \( \P \) gegeben durch die Wahrscheinlichkeiten \( p_n = \P[\{\omega_n\}] \). Das bedingte W'mass \( \P^*[ \ \cdot \ ] = \P[ \ \cdot \ | B] \) ist dann gegeben durch
\[
    p_n^* = \P^*[\{\omega_n\}] = \P[\{\omega_n\} |B] = \begin{cases}
        \frac{p_n}{\P[B]} & \text{für \( \omega_n \in B \),} \\
        0 & \text{für \( \omega_n \notin B \),}
    \end{cases}
\]

\begin{mainbox}{Satz von der totalen Wahrscheinlichkeit}
    Sei \( B_1, \dots, B_N \) mit \( \P[B_n] > 0 \) für jedes \( 1 \leq n < N \) eine \emph{Partition} des Grundraums \( \Omega \). Dann gilt für alle \( A \in \F \)
    \[
        \P[A] = \sum_{n = 1}^N \P[A \: | \: B_n] \, \P[B_n]
    \]
\end{mainbox}

\begin{mainbox}{Satz von Bayes}
    Sei \( B_1, \dots, B_N \in \F \) eine Partition von \( \Omega \) mit \( \P[B_n] > 0 \) für alle \( n \). Für jedes \( A \) mit \( \P[A] > 0 \) und jedes \( n \in \{1, \dots, N\} \) gilt
    \[
        \P[B_n | A] = \frac{\P[A \: | \: B_n] \, \P[B_n]}{\sum_{k = 1}^N \P[A \: | \: B_k] \, \P[B_k]}
    \]

    \begin{subbox}{Spezialfall \( n = 2 \)}
        Haben wir eine Partition \( \Omega = B \cup B^\complement \) gilt:
        \[
            \P[B \: | \: A] = \frac{\P[A \: | \: B] \, \P[B]}{\P[A \: | \: B] \, \P[B] + \P[A \: | \: B^\complement] \, \P[B^\complement]}
        \]
    \end{subbox}
\end{mainbox}

\subsection{Unabhängigkeit}

\begin{mainbox}{Unabhängigkeit zweier Ereignisse}
    Sei \(\Omega, \F, \P) \) ein W'raum. Zwei Ereignisse \( A \) und \( B \) heissen \emph{Unabhängig}, falls
    \[
        \P[A \cap B] = \P[A] \, \P[B]
    \]
\end{mainbox}
\begin{itemize}
    \item Falls \( \P[A] \in \{0, 1\} \), dann ist \( A \) unabhängig von jedem Ereignis.
    \item Falls \( A \) unabhängig von sich selbst ist muss \( \P[A] \in \{0, 1\} \) gelten.
    \item \( A \) ist unabhängig von \( B \) \( \iff \) \( A \) unabhängig von \( B^\complement \).
\end{itemize}

\begin{subbox}{}
    Seien \( A, B \in \F \) zwei Ereignisse mit \( \P[A], \P[B] > 0 \). Dann sind die folgenden Aussagen äquivalent:
    \begin{enumerate}[label=(\roman*)]
        \item \( \P[A \cap B] = \P[A] \, \P[B] \), \emph{\( A \) und \( B \) sind unabhängig},
        \item \( \P[A \: | \: B] = \P[A] \), \emph{Eintreten von \( B \) hat keinen Einfluss auf \( A \)},
        % \item \( \P[B \: | \: A] = \P[B] \), \emph{Eintreten von \( A \) hat keinen Einfluss auf \( B \)}.
    \end{enumerate}
\end{subbox}

\begin{mainbox}{Unabhängigkeit}
    Sei \( I \) eine beliebige Indexmenge. Eine Familie von Ereignissen \( (A_i)_{i \in I} \) heisst \emph{unabhängig}, wenn für alle endlichen Teilmengen \( J \subset I \) gilt:
    \[
        \P \left[ \bigcap_{j \in J} A_j \right] = \prod_{j \in J} \P[A_j]
    \]
\end{mainbox}

\begin{subbox}{Paarweise Unabhängigkeit}
    Hat man die Produktformel nur für alle Paare von Ereignissen, heissen diese \emph{paarweise unabhängig}. Unabhängigkeit \( \implies \) paarweise Unabhängigkeit.
\end{subbox}

\begin{mainbox}{Eigenschaften von Verteilungsfunktionen}
    Sei \( X \) eine ZV auf \( (\Omega, \F, \P) \). \( F_X: \: \R \to [0, 1] \) erfüllt:
    \begin{enumerate}[label=(\roman*)]
        \item \( F_X \) ist \emph{monoton wachsend}.
        \item \( F_X \) ist \emph{rechtsstetig}, d.h. \( \forall x \in \R: \: F_X(x) = \lim_{h \searrow 0} F_X(x + h) \).
        \item \( \lim_{x \to -\infty} F_X(X) = 0 \) und \( \lim_{x \to \infty} F_X(x) = 1 \).
    \end{enumerate}
\end{mainbox}

\subsubsection{Gemeinsame Verteilungsfunktion}

\begin{mainbox}{Gemeinsame Verteilungsfunktion}
    Seien \( X_1, \dots, X_n \) ZV. Die \emph{gemeinsame Verteilungsfunktion von \( X_1, \dots, X_n \)} ist die Abbildung \( F: \: \R^n \to [0, 1] \) definiert durch
    \begin{multline*}
        (x_1, \dots, x_n) \mapsto F(x_1, \dots, x_n) \\ = \P[X_1 \leq x_1, \, \dots, \, X_n \leq x_n]
    \end{multline*}
\end{mainbox}

\subsection{Unabhängigkeit von ZV}

\begin{mainbox}{Unabhängigkeit}
    Seien \( X_1, \dots, X_n \) ZV auf \( (\Omega, \F, \P) \). Dann heissen \( X_1, \dots, X_n \) \emph{unabhängig}, wenn für alle \( x_1, \dots, x_n \in \R \) gilt:
    \[
        \P[X_1 \leq x_1, \, \dots, \, X_n \leq x_n] = \P[X_1 \leq x_1] \cdot \dots \cdot \P[X_n \leq x_n]
    \]
\end{mainbox}
\paragraph{Bemerkung} \( X_1, \dots, X_n \) unabhängig \( \iff \) für alle Intervalle \( I_1 \subset \R, \, \dots, \, I_n \subset \R \) sind die Ereignisse \( \{X_1 \in I_1\}, \, \dots, \, \{X_n \in I_n\} \) unabhängig.

\subsubsection{Gruppierungen}

Wenn wir eine Menge unabhängiger ZV haben und disjunkte Gruppen solcher ZV bilden, dann sind diese Gruppen auch wiederum unabhängig voneinander.

\begin{mainbox}{Gruppierung von ZV}
    Seien \( X_1, \dots, X_n \) unabhängige ZV. Seien \( 1 \leq i_1 < i_2 < \dots < i_k \leq n \) Indexe und \( \varphi_1, \dots, \varphi_k \) Abbildungen. Dann sind \( Y_1 \coloneqq \varphi_1(X_1, \dots, X_{i_1}), \; Y_2 \coloneqq \varphi_2(X_1, \dots, X_{i_2}), \; \dots, \; Y_k \coloneqq \varphi_k(X_1, \dots, X_{i_k}) \) unabhängig.
\end{mainbox}

\subsubsection{i.i.d.}

\begin{mainbox}{Unabhängig und identisch verteilt}
    Eine Folge \( X_1, X_2, \dots\) heisst
    \begin{itemize}
        \item \emph{unabhängig} falls \( X_1, \dots, X_n \) für alle \( n \in \N \) unabhängig sind,
        \item \emph{unabhängig und identisch verteilt}, falls sie unabhängig ist und die ZV dieselbe Verteilungsfunktion haben.
    \end{itemize}
\end{mainbox}

\subsection{Konstruktion von ZV}

Unser Vorgehen ist in 4 Schritte unterteilt:
\begin{enumerate}
    \item Existenzsatz von Kolmogorov und Folgen von i.i.d. Bernoulli-ZV
    \item Konstruktion von gleichverteilten ZV auf \( [0, 1] \)
    \item Konstruktion von ZV mit beliebiger Verteilungsfunktion \( F \)
    \item allgemeine Folgen von unabhängigen ZV
\end{enumerate}

\subsubsection{Existenzsatz von Kolmogorov}

\begin{mainbox}{Existenzsatz von Kolmogorov}
    Es existiert ein W'raum \( (\Omega, \F, \P) \) und eine unendliche Folge von i.i.d. Bernoulli-ZV \( X_1, X_2, \dots \) auf \( (\Omega, \F, \P) \) mit Parameter \( \frac{1}{2} \).
\end{mainbox}

\subsubsection{Konstruktion gleichverteilter ZV auf \( [0, 1] \)}

% Wir verwenden nun Bernoulli-ZV, um eine \emph{(stetige) gleichverteilte ZV auf dem Intervall \( [0, 1] \)} zu konstruieren.

% \begin{mainbox}{}
%     Eine ZV \( U \) heisst \emph{gleichverteilt auf \( [0, 1] \)}, wir schreiben \( U \sim \uniform([0, 1]) \), falls ihre Verteilungsfunktion gegeben ist durch
%     \[
%         F_U(x) = \begin{cases}
%             0 & x < 0 \\
%             x & 0 \leq x \leq 1 \\
%             1 & x > 1
%         \end{cases}
%     \]
% \end{mainbox}

Sei \( X_1, X_2, \dots \) eine Folge von unabhängigen Bernoulli-ZV mit Parameter \( \frac{1}{2} \). Für jedes fixe \( \omega \in \Omega \) gilt \( X_1(\omega), \, X_2(\omega), \, \dots \in \{0, 1\} \). Somit konvergiert die unendliche Reihe
\begin{equation}
    X(\omega) \coloneqq \sum_{n = 1}^\infty 2^{-n} X_n(\omega)
    \label{eq:slides_2_3}
\end{equation}
absolut, und es gilt \( X(\omega) \in [0, 1] \).

\begin{subbox}{}
    Die Abbildung \( X: \: \Omega \to [0, 1] \) in Gleichung~\eqref{eq:slides_2_3} ist eine gleichverteilte ZV auf \( [0, 1] \).
\end{subbox}

\subsubsection{Konstruktion mit beliebiger Ver\-tei\-lungs\-funk\-tion \( F \)}

% Sei \( F: \R \to [0, 1] \) eine Funktion, die die Eigenschaften (i)–(iii) von Verteilungsfunktionen erfüllt.
% Falls \( F \) streng monoton steigend und stetig ist, dann ist \( F \) bijektiv und es existiert eine Umkehrfunktion \( F^{-1} \). Für jedes \( \alpha \in [0, 1] \) ist \( x \coloneqq F^{-1}(\alpha) \) die eindeutige reelle Zahl, für die \( F(x) = \alpha \) gilt.

% Allgemeiner kann man die sogenannte \emph{verallgemeinerte inverse Verteilungsfunktion} oder \emph{Quantil-Funktion} für \( F \) definieren.

\begin{mainbox}{Verallgemeinerte inverse Verteilungsfunktion}
    Die \emph{verallgemeinerte inverse Verteilungsfunktion} von \( F \) ist \( F^{-1}: \: (0, 1) \to \R \) definiert durch
    \[
        F^{-1}(\alpha) = \inf\{x \in \R \: | \: F(x) \geq \alpha\}
    \]
    Nach Definition des Infimums und unter Verwendung der Rechtsstetigkeit von \( F \) gilt für jedes \( x \in \R \) und \( \alpha \in (0, 1) \), dass
    \[
        F^{-1}(\alpha) \leq x \iff \alpha \leq F(x)
    \]
\end{mainbox}

% Mithilfe der verallgemeinerten Verteilungsfunktion können wir nun ZV mit beliebigen Verteilungsfunktionen konstruieren.

\begin{mainbox}{Inversionsmethode}
    Sei \( F: \: \R \to [0, 1] \) eine Abbildung mit den Eigenschaften (i)–(iii) von Verteilungsfunktionen. Sei \( U \sim \uniform([0, 1]) \). Dann hat \( X \coloneqq F^{-1}(U) \) die Verteilungsfunktion \( F \).
\end{mainbox}

% Wir bemerken, dass \( X = F^{-1}(U) \) strenggenommen nur auf einer Menge mit Wahrscheinlichkeit 1 (da \( \P[U \in (0, 1)] = 1 \)) aber nicht unbedingt auf ganz \( \Omega \) definiert ist. Wir beheben das Problem mittels folgender Definition
\[
    X(\omega) = \begin{cases}
        F^{-1}(U(\omega)) & U(\omega) \in (0, 1) \\
        0 & \text{sonst}
    \end{cases}
\]
Dabei spielt \( 0 \) selbst keine Rolle, man kann jede beliebige reelle Zahl nehmen.

\subsubsection{Allgemeine Folgen unabhängiger ZV}

\begin{mainbox}{}
    Sei \( F_1, F_2, \dots \) eine Folge von Funktionen \( \R \to [0, 1] \), die die Eigenschaften (i)–(iii) von Verteilungsfunktionen erfüllen. Dann existiert ein W'raum \( (\Omega, \F, \P) \) und eine Folge von ZV \( X_1, X_2, \dots \) auf diesem W'raum, sodass
    \begin{itemize}
        \item für jedes \( k \) gilt \( X_k \) hat Verteilungsfunktion \( F_k \)
        \item \( X_1, X_2, \dots \) sind unabhängig
    \end{itemize}
    \label{satz:slides_2_35}
\end{mainbox}

Dieser Satz erlaubt es uns, direkt mit ZV zu arbeiten, ohne \( (\Omega, \F, \P) \) genauer zu definieren.

\section{Diskrete und stetige ZV}

\subsection{(Un)stetigkeit der Verteilungsfunktion}

\begin{mainbox}{Wahrscheinlichkeit eines Punktes}
    Sei \( X: \: \Omega \to \R \) eine ZV mit Verteilungsfunktion \( F \). Für jedes \( x \in \R \) gilt
    \[
        \P[X = x] = F(x) - F(x-)
    \]
\end{mainbox}

\paragraph{Interpretation} Sei \( x \in \R \) fixiert.
\[
    \P[X = x] = \begin{cases}
        \text{\say{Sprunghöhe}} \ F(x) - F(x-) & F \ \text{nicht stetig in} \ x \\
        0 & F \ \text{stetig in} \ x
    \end{cases}
\]

\subsection{F.s. Ereignisse}

\begin{mainbox}{}
    Sei \( A \in \F \) ein Ereignis. Wir sagen \emph{\( A \) tritt \( \P \)-f.s. ein}, falls \( \P[A] = 1 \).
    Wenn klar ist, welches W'mass \( \P \) gemeint ist, kürzen wir ab und schreiben einfach \emph{f.s.}.
\end{mainbox}

Auch für allgemeine Mengen \( A \subset \Omega \) (nicht unbedingt \( A \in \F \). \( A \) f.s. \( \impliedby \) \( A' \in \F \) existiert, sodass \( A' \subset A \) und \( \P[A'] = 1 \).

\subsection{Diskrete ZV}

% \begin{mainbox}{Diskrete ZV}
%     Eine ZV \( X: \: \Omega \to \R \) heisst \emph{diskret}, falls eine endliche oder abzählbare Menge \( W \subset \R \) existiert, sodass \( \P[X \in W] = 1 \), wenn also die Werte von \( X \) fast sicher in \( W \) liegen.
% \end{mainbox}

% Wenn der Grundraum \( \Omega \) endlich oder abzählbar ist, dann ist jede ZV \( X: \: \Omega \to \R \) diskret. In der Tat ist das Bild
% \[
%     X(\Omega) = \{x \in \R \: | \: \exists \omega \in \Omega: \: X(\omega) = x\}
% \]
% endlich oder abzählbar und wir haben \( \P[X \in W] = 1 \) mit \( W = X(\Omega) \).

\begin{mainbox}{Gewichtsfunktion}
    Für diskrete ZV \( X \) mit Wertebereich \( W(X) = \{ x_1, x_2, \dots \} \) und den Wahrscheinlichkeiten \( \{p_1, p_2, \dots \} \) ist die \emph{Gewichtsfunktion} von \( X \):
    \[
        p_X: \: W(X) \to [0, 1] \quad \text{mit} \quad p_X(x_k) \coloneqq \P[X = x_k] = p_k
    \]
    \( \{p_X(x_k)\}_{x_k \in W(X)} \) nennen wir auch \emph{Verteilung von \( X \)}.
\end{mainbox}

\begin{subbox}{}
    \( p_X \) hat folgende Eigenschaften:
    \begin{itemize}
        \item Für alle \( x_k \in W(X) \) gilt \( p_X(x_k) \in [0, 1] \).
        \item Die Wahrscheinlichkeiten addieren sich zu \( 1 \)
        \[
            \sum_{x_k \in W(X)} p_X(x_k) = \P[X \in W(X)] = 1
        \]
    \end{itemize}
\end{subbox}

% Umgekehrt, wenn wir eine Folge von Zahlen \( (p(x))_{x \in W} \) mit Werten in \( [0, 1] \) haben, sodass
% \[
%     \sum_{x \in W} p(x) = 1
% \]
% dann gibt es nach dem Satz in \ref{satz:slides_2_35} einen W'raum \( (\Omega, \F, \P) \) und eine ZV \( X \) mit zugehöriger Verteilung \( (p(x))_{x \in W} \).
% Diese Beobachtung ist in der Praxis wichtig, denn sie erlaubt uns zu schreiben: \say{Sei \( X \) eine diskrete ZV mit Verteilung \( (p(x))_{x \in W} \).}

% Der Zusammenhang zwischen Gewichtsfunktion und Verteilungsfunktion ist im folgenden Satz beschrieben.

\begin{mainbox}{Verteilungsfunktion}
    Sei \( X \) eine diskrete ZV mit Werten in \( W \) und Gewichtsfunktion \( p_X \). Dann ist die Verteilungsfunktion von \( X \) gegeben durch
    \begin{equation}
        F_X(x) = \P[X \leq x] = \sum_{\substack{y \leq x \\ y \in W}} p_X(y)
        \label{eq:distribution_given_density}
    \end{equation}
\end{mainbox}

% Insbesondere ist \( F_X \) also durch \( p_X \) \emph{vollständig festgelegt}. Mit dem gleichen Argument erhält man auch für jede messbare Teilmenge \( B \subseteq W \)
% \[
%     \P[X \in B] = \sum _{x \in B} p_X(x)
% \]

% Gleichung~\eqref{eq:distribution_given_density} drückt die Verteilungsfunktion \( F_X \) in Bezug auf \( p_X \) als eine stückweise konstante Funktion aus.
% Umgekehrt ist eine ZV mit einer stückweisen konstanten Verteilungsfunktion \( F_X \) diskret. \( W \) und \( p_X \) sind dann gegeben durch
\begin{align*}
    W &= \{\text{Sprungstellen von } F_X\} \\
    p(x) &= \text{\say{Sprunghöhe} im Punkt } x \in W
\end{align*}

% \subsubsection{Beispiele diskreter ZV}

% \paragraph{Bernoulli-Verteilung (\( X \sim \Ber(p) \))}
% \[
%     \P[X = 0] = 1 - p \quad \text{und} \quad \P[X = 1] = p
% \]

\subsubsection{Binomialverteilung (\( X \sim \Bin(n, p) \))}
% Modelliert \emph{Anzahl Erfolge} bei wiederholten Bernoulli-Experimenten.
% \[
%     \P[X = k] = \binom{n}{k} \, p^k \, (1-p)^{n - k}
% \]

\begin{subbox}{Summe unabhängiger Bernoulli-ZV}
    Sei \( p \in [0, 1] \) und \( n \in \N \). Seien \( X_1, \dots, X_n \sim \Ber(p) \) und alle unabhängig. Dann gilt
    \[
        S_n \coloneqq X_1 + \dots + X_n \sim \Bin(n, p)
    \]
\end{subbox}

\subsubsection{Geometrische Verteilung (\( X \sim \Geom(p) \))}
% Beschreibt \emph{Wartezeit bis zum nächsten Erfolg} in einer unendlichen Folge von Bernoulli-Experimenten.
% \[
%     \P[X = k] = p(1 - p)^{k - 1}
% \]

\begin{subbox}{}
    Seien \( X_1, X_2, \dots \sim \Ber(p) \) unabhängig. Dann ist
    \[
        T \coloneqq \inf\{n \geq 1 \: | \: X_n = 1 \}
    \]
    eine geometrisch verteilte ZV mit Parameter \( p \).
\end{subbox}

Sei \( T \sim \Geom(p) \), dann ist \( T > n \), wenn die ersten \( n \) Bernoulli-Experimente fehlschlagen. Daher gilt
\[
    \P[T > n] = (1 - p)^n
\]

\begin{subbox}{\say{Gedächtnislosigkeit} der geometrischen Verteilung}
    Sei \( T \sim \Geom(p) \) mit \( p \in (0, 1) \). Dann gilt für alle \( n \geq 0 \) und alle \( k \geq 1 \)
    \[
        \P[T \geq n + k \: | \: T > n] = \P[T \geq k]
    \]
\end{subbox}

\subsubsection{Negativbinomiale Verteilung (\( X \sim \NBin(r, p) \))}
% Beschreibt \emph{Wartezeit bis zum \( r \)-ten Erfolt} in einer unendlichen Folge von Bernoulli-Experimenten.
% \[
%     \P[X = k] = \binom{k - 1}{r - 1} \, p^r \, (1 - p)^{k - r}
% \]

\begin{subbox}{}
    Seien \( X_1, X_2, \dots \sim \Ber(p) \) unabhängig. Dann hat
    \[
        T_r \coloneqq \inf\left\{ n \geq 1 \: \middle| \: \sum_{l = 1}^n X_l = r \right\}
    \]
    eine negativbinomiale Verteilung mit Paramentern \( r \) und \( p \).
\end{subbox}

Sind \( X_1, \dots, X_r \sim \Geom(p) \) und unabhängig, so ist ihre Summe \( X = X_1 + \dots + X_r \sim \NBin(r, p) \).

\subsubsection{Hypergeometrische Verteilung (\( X \sim \hypergeom(n, r, m) \))}
% \[
%     \P[X = k] = \frac{\binom{r}{k} \binom{n - r}{m - k}}{\binom{n}{m}}
% \]

\begin{subbox}{}
    Seien in einer Urne \( n \) Gegenstände, davon \( r \) vom Typ 1 und \( n - r \) vom Typ 2. Man ziehe \( m \) der Gegenstände ohne Zurücklegen. Sei \( X \) nun die Anzahl der gezogenen Gegenstände vom Typ 1. Dann ist \( X \) hypergeometrisch mit den Parametern von oben verteilt.
\end{subbox}

\subsubsection{Poisson-Verteilung (\( X \sim \Poisson(\lambda) \))}
% \[
%     \P[X = k] = \frac{\lambda^k}{k!} \, e^{-\lambda}
% \]

\begin{subbox}{Poisson-Approximation der Binomialverteilung}
    Sei \( \lambda > 0 \). Für jedes \( n \geq 1 \) betrachten wir \( n \) ZV \( X_n \sim \Bin(n, \lambda) \). Sei \( N \sim \Poisson(\lambda) \). Dann gilt für alle \( k \in \N \)
    \[
        \lim_{n \to \infty} \P[X_n = k] = \P[N = k]
    \]
\end{subbox}

Siehe \emph{Konvergenz in Verteilung} in  \ref{box:konvergenz_in_verteilung}.

\subsection{Stetige ZV}

\begin{mainbox}{Stetig verteilte ZV}
    \( X: \: \Omega \to \R \) heisst \emph{stetig}, wenn \( f_X: \: \R \to \R_+ \) existiert, sodass \( F_X \) dargestellt werden kann als
    \[
        F_X(x) = \int_{-\infty}^x f_X(t) \, \mathrm{d}t \text{.}

\end{mainbox}

% \]

% Ein wichtiger Spezialfall ist die \emph{Standardnormalverteilung} mit \( \mu = 0 \) und \( \sigma^2 = 1 \), also \( \normal(0, 1) \).
% Die zugehörige Dichte wird meistens mit \( \varphi \) und die Verteilungsfunktion mit \( \Phi \) bezeichnet. Die dazugehörige Variable nennen wir meistens \( Z \).

\begin{subbox}{Standardnormalverteilung}
    Für \( X \sim \normal(\mu, \sigma^2) \) gilt \( \frac{X - \mu}{\sigma} \sim \normal(0, 1) \), also
    \[
        F_X(x) = \P[X \leq x] = \P\left[\frac{X - \mu}{\sigma} \leq \frac{x - \mu}{\sigma}\right] = \Phi\left(\frac{x - \mu}{\sigma}\right)
    \]
\end{subbox}

Seien \( X_1, \dots, X_n \) unabhängige normalverteilte ZV mit Parametern \( (\mu_1, \sigma_1^2), \dots, (\mu_n, \sigma_n^2) \). Dann gilt
\[
    Z \coloneqq \mu_0 + \sum_{k = 1}^n a_k X_k \sim \normal\left( \mu_0 + \sum_{k = 1}^n a_k \mu_k, \, \sum_{k = 1}^n  a_k^2 \sigma_k^2 \right)
\]

Für \( \mu \in \R \), \( \sigma^2 > 0 \), und \( Z \sim \normal(0, 1) \) gilt
\[
    \mu + \sigma Z \sim \normal(\mu, \sigma^2)
\]

\section{Erwartungswert, Varianz und Kovarianz}

\subsection{Allgemeiner Erwartungswert}

\begin{subbox}{Erwartungswert (nicht-negativ)}
    Sei \( X: \: \Omega \to \R_+ \) eine ZV mit \emph{nicht-negativen Werten}.
    Dann heisst
    \[
        \E[X] = \int_0^\infty (1 - F_X(x)) \, \mathrm{d}x
    \]
    der \emph{Erwartungswert von X}.
\end{subbox}
Der Erwartungswert ist immer definiert und kann sowohl endlich als auch unendlich werden.

\begin{mainbox}{}
    Sei \( X \) eine nicht-negative ZV. dann gilt \( \E[X] \geq 0 \).
    Gleichheit gilt genau dann wenn \( X = 0 \) fast sicher gilt.
\end{mainbox}

Für allgemeine ZV definieren wir den Erwartungswert durch \emph{Zerlegung in einen positiven und einen negativen Teil}
\[
    X_+ = \max(X, 0) \quad \text{und} \quad X_- = -\min(X, 0) = \max(-X, 0)
\]

Sowohl \( X_+ \) als auch \( X_- \) sind \emph{nicht-negative ZV}.
Zudem gilt
\[
    X = X_+ - X_- \quad \text{und} \quad |X| = X_+ + X_-
\]

\begin{mainbox}{Allgemeiner Erwartungswert}
    Sei X eine reellwertige ZV. Falls \( \E[|X|] < \infty \) dann heisst
    \[
        \E[X] = \E[X_+] - \E[X_-]
    \]
    der \emph{Erwartungswert von \( X \)}.
\end{mainbox}

\begin{itemize}
    \item \( \E[|X|] < \infty \implies \E[X_-], \E[X_+] < \infty \), da \( |X| = X_+ + X_- \).
    \item Für \( X \geq 0 \) ist \( \E[X] \) immer definiert. Er kann endlich oder unendlich sein.
    \item Wenn \( X \) kein konstantes Vorzeichen hat und \( \E[|X|] < \infty \) nicht erfüllt ist, ist \( \E[X] \) \emph{undefiniert}.
\end{itemize}

\subsection{Erwartungswert diskreter ZV}

\begin{mainbox}{Erwartungswert (diskret)}
    Sei \( X: \: \Omega \to \R \) eine diskrete ZV, deren Werte fast sicher in \( W \) (endlich oder abzählbar) liegen. Dann gilt
    \[
        \E[X] = \sum_{x \in W} x \cdot \P[X = x] = \sum_{x \in W} x \cdot p_X(x)
    \]
    solange der Erwartungswert wohldefiniert ist.
\end{mainbox}
Der Erwartungswert ist wohldefiniert, sofern die Reihe \( \sum_{x \in W} x \cdot \P[X = x] \) absolut konvergiert, d.h. falls
\[
    \sum_{x \in W} |x| \cdot \P[X = x] < \infty
\]
Ist die Reihe nicht absolut konvergent, so existiert der Erwartungswert nicht.

% \paragraph{Bernoulli (\( X \sim \Ber(p) \))} \( \E[X] = p \).

% \paragraph{Poisson (\( X \sim \Poisson(\lambda) \))} \( \E[X] = \lambda \).

% \paragraph{Binomialverteilung (\( X \sim \Bin(n, p) \))} \( \E[X] = np \)

\subsubsection{Erwartungswerte transformierter ZV}

\begin{mainbox}{}
    Seien \( X: \: \Omega \to \R \) eine diskrete ZV und \( \varphi: \: \R \to \R \). Dann gilt
    \[
        \E[\varphi(X)] = \sum_{x \in W} \varphi(x) \cdot \P[X = x]
    \]
    solange der Erwartungswert wohldefiniert ist.
\end{mainbox}

\subsection{Erwarungswert stetiger ZV}

\begin{mainbox}{Erwartungswert (stetig)}
    Sei \( X \) eine stetige ZV. Dann gilt
    \[
        \E[X] = \int_{-\infty}^\infty x f_X(x) \, \mathrm{d}x
    \]
    solange das Integral absolut konvergiert, d.h. falls \( \int_{-\infty}^\infty |x| f_X(x) \, \mathrm{d}x < \infty \). Ist das Integral nicht absolut konvergent, so existiert der Erwartungswert nicht.
\end{mainbox}

\begin{mainbox}{}
    Seien \( X \) eine stetige ZV und \( \varphi: \: \R \to \R \). Dann gilt
    \[
        \E[\varphi(X)] = \int_{-\infty}^\infty \varphi(x) f_X(x) \, \mathrm{d}x
    \]
    solange das Integral wohldefiniert ist.
\end{mainbox}

% \paragraph{Gleichverteilung (\( X \sim \uniform([a, b] \))} \( \E[X] = \frac{a + b}{2} \)

% \paragraph{Exponentialverteilung (\( X \sim \Exp(\lambda) \))} \( \E[X] = \frac{1}{\lambda} \)

\subsubsection{Normalverteilung (\( X \sim \normal(\mu, \sigma^2) \))}
% \( \E[X] = \mu \)

\begin{subbox}{}
    Für das uneigentliche Integral über die \emph{gausssche Glockenkurve} gilt
    \[
        \int_{-\infty}^\infty e^{\frac{-x^2}{2 \sigma^2}} \, \mathrm{d}x = \sqrt{2 \pi \sigma^2}
    \]
\end{subbox}

\subsubsection{Cauchy-Verteilung}
Für die Cauchy-Verteilung \emph{existiert kein Erwartungswert}.

\subsection{Eigenschaften des Erwartungswertes}

\begin{mainbox}{Linearität des Erwartunswertes}
    Seien \( X, Y: \: \Omega \to \R \) ZV und sei \( \lambda \in \R \). Falls die Erwartungswerte wohldefiniert sind, gilt:
    \begin{enumerate}
        \item \( \E[\lambda X] = \lambda \E[X] \)
        \item \( \E[X + Y] = \E[X] + \E[Y] \)
    \end{enumerate}
\end{mainbox}

\begin{mainbox}{Monotonie des Erwartungswertes}
    Seien \( X, Y \) zwei ZV, sodass \( X \leq Y \) f.s. gilt. Falls beide Erwartungswerte wohldefiniert sind, gilt:
    \[
        \E[X] \leq \E[Y]
    \]
\end{mainbox}

\begin{mainbox}{Erwartungswerte bei Unabhängigkeit}
    Seien \( X, Y \) zwei unabhängige ZV mit endlichen Erwartungswerten. Dann gilt:
    \[
        \E[XY] = \E[X] \E[Y]
    \]
\end{mainbox}

\subsubsection{Charakterisierung mittels Testfunktion: Dichte}

\begin{mainbox}{}
    Sei \( X \) eine ZV und sei \( f: \: \R \to \R_+ \) eine Abbildung, sodass \( \int_{-\infty}^{\infty} f(x) \, \mathrm{d}x = 1 \). Dann sind folgende Aussagen äquivalent
    \begin{enumerate}
        \item \( X \) ist stetig mit Dichte \( f \)
        \item für jede stückweise stetige, beschränkte Abbildung \( \varphi: \: \R \to \R \) gilt
        \[
            \E[\varphi(X)] = \int_{-\infty}^\infty \varphi(x) f(x) \, \mathrm{d}x
        \]
    \end{enumerate}
\end{mainbox}

\subsubsection{Charakterisierung mittels Testfunktion: Unabhängigkeit}

\begin{mainbox}{}
    Seien \( X, Y \) zwei ZV. Die folgenden Aussagen sind äquivalent:
    \begin{enumerate}
        \item \( X, Y \) sind unabhängig
        \item Für alle stücksweise stetigen, beschränkten Abbildungen \( \varphi, \psi: \: \R \to \R \) gilt
        \[
            \E[\varphi(X) \psi(Y)] = \E[\varphi(X)] \E[\psi(Y)]
        \]
    \end{enumerate}
\end{mainbox}

Die gleiche Aussage gilt ebenfalls für \( n \) ZV \( X_1, \dots, X_n \).
\begin{mainbox}{}
    Seien \( X_1, \dots, X_n \) ZV. Die folgenden Aussagen sind äquivalent:
    \begin{enumerate}
        \item \( X_1, \dots, X_n \) sind unabhängig
        \item Für alle stückweise stetigen, beschränkten Abbildungen \( \varphi_1, \dots, \varphi_n: \: \R \to \R \) gilt
        \[
            \E[\varphi_1(X_1) \cdot \dots \cdot \varphi_n(X_n)] = \E[\varphi_1(X_1)] \cdot \dots \cdot \E[\varphi_n(X_n)]
        \]
    \end{enumerate}
\end{mainbox}

\subsection{Ungleichungen}

\begin{mainbox}{Markow-Ungleichung}
    Sei \( X \) eine nicht-negative ZV und sei \( g: \: X(\Omega) \to [0, \infty) \) eine wachsende Funktion. Für jedes \( c \in \R \) mit \( g(c) > 0 \) gilt
    \[
        \P[X \geq c] \leq \frac{\E[g(X)]}{g(c)}
    \]
\end{mainbox}

\begin{subbox}{Chebychev-Ungleichung}
    Sei \( Y \) eine ZV mit endlicher Varianz. Für jedes \( c > 0 \) gilt dann
    \[
        \P[|Y - \E[Y]| \geq c] \leq \frac{\V[Y]}{c^2}
    \]
\end{subbox}

\begin{mainbox}{Jensensche Ungleichung}
    Sei \( X \) eine ZV und sei \( \varphi: \: \R \to \R \) eine konvexe Funktion. Falls \( \E[\varphi(X)] \) und \( \E[X] \) wohldefiniert sind, gilt
    \[
        \varphi(\E[X]) \leq \E[\varphi(X)]
    \]
\end{mainbox}

\begin{subbox}{Dreiecksungleichung}
    Anwendung der Jensenschen Ungleichung auf \( \varphi(x) = |x| \) liefert die \emph{Dreiecksungleichung}
    \[
        |\E[X]| \leq \E[|X|]
    \]
\end{subbox}

Die Jensensche Ungleichung liefert für die konvexe Funktion \( \varphi(x) = x^2 \)
\[
    \E[|X|] \leq \sqrt{\E[X^2]} \iff \E[X]^2 \leq \E[X^2]
\]

\subsection{Varianz}

\begin{mainbox}{Varianz}
    Sei \( X \) eine ZV, sodass \( \E[X^2] < \infty \). Die \emph{Varianz von \( X \)} ist:
    \[
        \V[X] = \E[(X - \E[X])^2] = \E[X^2] - \E[X]^2
    \]

    \( \sigma = \sigma_x = \sigma(X) = \sqrt{\V[X]} \) heisst \emph{Standardabweichung von \( X \)}.
\end{mainbox}

\begin{itemize}
    \item Falls \( \E[X^2] < \infty \), dann gilt laut der Dreiecksungleichung auch \( \E[|X|] < \infty \) und insbesondere \( \E[X] < \infty \).
    % \item Die Varianz bzw. die Standardabweichung ist ein \emph{Indikator für die Fluktuation von \( X \) um den Mittelwert \( \E[X] \)} herum.
\end{itemize}

Es gilt
\begin{itemize}
    \item \( \V[X] \geq 0 \), und Gleichheit genau dann, wenn \( X = a 1_\Omega \) eine Konstante ist.
    \item \( \V[aX] = a^2 \V[X] \).
    \item \( \V[X + a] = \V[X] \).
\end{itemize}

\begin{subbox}{}
    Im Allgemeinen gilt \( \V\left[ \sum_{k = 1}^n X_k \right] \neq \sum_{k = 1}^n \V[X_n] \). Bei Unabhängigkeit erhalten wir allerdings Gleichheit.

    Seien \( X_1, \dots, X_n \) paarweise unabhänge ZV, dann gilt
    \[
        \V\left[ \sum_{k = 1}^n X_k \right] = \sum_{k = 1}^n \V[X_k]
    \]
\end{subbox}

% \paragraph{Bernoulli (\( X \sim \Ber(p) \))} \( \V[X] = p(1 - p) \)

% \paragraph{Poisson (\( X \sim \Poisson(\lambda) \))} \( \V[X] = \lambda \)

% \paragraph{Gleichverteilung (\( X \sim \uniform([a, b]) \))} \( \V[X] = \frac{(b - a)^2}{12} \)

% \paragraph{Binomialverteilung (\( S \sim \Bin(n, p) \))} \( \V[S] = np(1 - p) \)

% \paragraph{Normalverteilung (\( X \sim \normal(\mu, \sigma^2) \))} \( \V[X] = \sigma^2 \)

\subsection{Kovarianz}

\begin{mainbox}{Kovarianz}
    Seien \( X, Y \) zwei ZV mit endlichen zweiten Momenten \( \E[X^2], \E[Y^2] < \infty \). Die \emph{Kovarianz zwischen \( X \) und \( Y \)} ist definiert als
    \[
        \cov(X, Y) = \E\big[(X - \E[X])(Y - \E[Y])\big] = \E[XY] - \E[X]\E[Y]
    \]
\end{mainbox}

% Die Wohldefiniertheit der Kovarianz ist durch die endlichen zweiten Momente sichergestellt. Das sieht man anhand der elementaren Ungleichung
% \[
%     |XY| \leq \frac{1}{2} X^2 + \frac{1}{2} Y^2
% \]
% Daraus folgt nämlich mit Monotonie und Linearität des Erwartungswerts folgende Ungleichung
% \[
%     \E[XY] \leq \frac{1}{2} \E[X^2] + \frac{1}{2} \E[Y^2] < \infty
% \]

Es gilt \( \cov(X, X) = \V[X] \).

Es gilt \(X, Y \ \text{sind unabhängig} \implies \cov(X, Y) = 0 \).
Die umgekehrte Implikation ist falsch! Aber es gilt \( X, Y \ \text{sind unabhängig} \iff \\ \forall \varphi, \psi \ \text{stückweise stetig, beschränkt} \ \: \cov(\varphi(X), \psi(Y)) = 0 \).

Wir verwenden folgende Terminologie:
\begin{itemize}
    \item Wenn \( \cov(X, Y) > 0 \), dann sind \( X \) und \( Y \) \emph{positiv korreliert}.
    \item Wenn \( \cov(X, Y) = 0 \), dann sind \( X \) und \( Y \) \emph{unkorrelliert}.
    \item Wenn \( \cov(X, Y) < 0 \), dann sind \( X \) und \( Y \) \emph{negativ korreliert} oder \emph{antikorreliert}.
\end{itemize}

Es gelten folgende Eigenschaften der Kovarianz für ZV \( X, Y, Z \) mit endlichem zweiten Moment und Konstanten \( a, b, c, d, e, f, g, h \in \R \).
\begin{itemize}
    \item \emph{Positive Semidefinitheit}: \( \cov(X, Y) \geq 0 \)
    \item \emph{Symmetrie}: \( \cov(X, Y) = \cov(Y, X) \)
    \item \emph{Bilinearität}: \( \cov(aX + b, cY + d) = ac \cov(X, Y) \) und \( \cov\big(X, (eY + f) + (gZ + h)\big) = e \cov(X, Y) + g \cov(X, Z) \)
\end{itemize}

Die Kovarianzen einer Folge \( X_1, \dots, X_n \) werden oft in der \emph{Kovarianzmaatrix} zusammengefasst:
\[
    \Sigma = \cov(\mathbf{X}) \\ = \begin{pmatrix} \cov(X_i, X_j) \end{pmatrix}_{i, j}
\]
% \begin{multline*}
%     \Sigma = \cov(\mathbf{X}) \\ = \begin{pmatrix}
%         \V[X_1] & \cov(X_1, X_2) & \dots & \cov(X_1, X_n) \\
%         \cov(X_2, X_1) & \V[X_2] & \dots & \cov(X_2, X_n) \\
%         \vdots & \vdots & \ddots & \vdots \\
%         \cov(X_n, X_1) & \cov(X_n, X_1) & \dots & \V[X_n] \\
%     \end{pmatrix}
% \end{multline*}

Für den Spezialfall von nur zwei ZV \( X, Y \) gilt:
\begin{gather*}
    \V[X + Y] = \V[X] + \V[Y] + 2 \cov(X, Y) \\
    \V[X - Y] = \V[X] + \V[Y] - 2 \cov(X, Y)
\end{gather*}

\section{Gemeinsame Verteilung}

\subsection{Gemeinsame diskrete Verteilung}

\begin{mainbox}{Gemeinsame diskrete Verteilung}
    Seien \( X_1, \dots, X_n \) diskrete ZV, seien für \( k \in \{1, \dots, n\} \) \( W_k \subset \R \) endlich oder abzählbar, sodass \( X_k \in W_k \) fast sicher gilt.
    Die \emph{gemeinsame Verteilung von \( (X_1, \dots, X_n) \)} ist eine Familie von Wahrscheinlichkeiten
    \[
        \{p(x_1, \dots, x_n)\}_{x_1 \in W_1, \dots, x_n \in W_n} \text{,}
    \]
    wobei \( p: \: \R^n \to [0, 1] \) die \emph{gemeinsame Gewichtsfunktion} ist:
    \[
        p(x_1, \dots, x_n) = \P[X_1 = x_1, \; \dots, \; X_n = x_n]
    \]
\end{mainbox}

\begin{mainbox}{}
    Eine gemeinsame Verteilung von \( X_1, \dots, X_n \) erfüllt stets
    \begin{equation}
        \sum_{x_1 \in W_1, \; \dots, \; x_n \in W_n} p(x_1, \dots, x_n) = 1
        \label{eq:slides_satz_5_3}
    \end{equation}
\end{mainbox}

Umgekehrt gibt es bei endlichen oder abzählbaren Familien von Mengen \( W_1, \dots, W_n \) und einer gegebenen Funktion \( p: \: W_1 \times \dots \times W_n \to [0, 1] \), die die Gleichung~\eqref{eq:slides_satz_5_3} erfüllt, einen W'raum und diskrete ZV mit Verteilung \( p \).

\begin{subbox}{}
    Aus der gemeinsamen Gewichtsfunktion \( p \) bekommt man die gemeinsame Verteilungsfunktion via
    \begin{multline*}
        F(x_1, \dots, x_n)
        = \P[X_1 \leq x_1, \; \dots, \; X_n \leq x_n] \\
        = \sum_{\mathclap{y_1 \leq x_1, \; \dots, \; y_n \leq x_n}} \P[X_1 = y_1, \; \dots, \; X_n = y_n ]
        = \sum_{\mathclap{y_1 \leq x_1, \; \dots, \; y_n \leq x_n}} p(y_1, \dots, y_n)
    \end{multline*}
\end{subbox}

\begin{mainbox}{Verteilung des Bildes}
    Seien \( n \geq 1 \), \( \varphi: \: \R^n \to \R \) und \( X_1, \dots, X_n \) diskrete ZV, mit Werten jeweils in \( W_1, \dots, W_n \) (f.s.). Dann ist \( Z = \varphi(X_1, \dots, X_n) \) eine diskrete ZV, die f.s. Werte in \( W = \varphi(W_1 \times \dots \times W_n) \) annimmt. Zudem ist die Verteilung von \( Z \) für alle \( z \in W \) gegeben durch
    \[
        \P[Z = z] = \sum_{\substack{x_1 \in W_1, \; \dots, x_n \in W_n \\ \varphi(x_1, \dots, x_n) = z}} \P[X_1 = x_1, \; \dots \; X_n = x_n]
    \]
\end{mainbox}

% Unter Kenntnis der Verteilung von \( X_1, \dots, X_n \) kann man die Verteilung der einzelnen \( X_k \) separat ermitteln. In diesem Zusammenhang wird die Verteilung von \( X_k \) als \emph{\( k \)-te Randverteilung} bezeichnet.

\begin{mainbox}{Randverteilung}
    Seien \( X_1, \dots, X_n \) diskrete ZV mit geminsamer Gewichtsfunktion \( p \). Für jedes \( k \in \{1, \dots, n\} \) und jedes \( x \in W_k \) gilt
    \[
        \P[X_k = x] = \sum_{\substack{x_l \in W_l \\ l \in \{1, \dots, n\} \setminus \{k\}}} p(x_1, \dots, x_{k - 1}, x, x_{k + 1}, \dots, x_n)
    \]
\end{mainbox}

Analog ist die Verteilungsfunktion der \( k \)-ten Randverteilung gegeben durch
\begin{multline*}
    F_{X_k}(x)
    = \P[X_k \leq x] \\
    = \P[X_1 < \infty, \; \dots, \; X_{k - 1} < \infty,
        X_k \leq x,
        X_{k + 1} < \infty, \; \dots, \; X_n < \infty] \\
    = \lim_{\substack{x_l \to \infty \\ l \in \{1, \dots, n\} \setminus \{k\}}} F(x_1, \dots, x_{k - 1}, x, x_{k + 1}, \dots, x_n)
\end{multline*}

\begin{mainbox}{Erwartungswert des Bildes}
    Seien \( X_1, \dots, X_n \) diskrete ZV mit gemeinsamer Verteilung \( \{p(x_1, \dots, x_n)\}_{x_1 \in W_1, \; \dots, \; x_n \in W_n} \) und sei \( \varphi: \: \R^n \to \R \). Es gilt:
    \[
        \E[\varphi(X_1, \dots, X_n)] = \sum_{x_1, \dots, x_n} \varphi(x_1, \dots, x_n) p(x_1, \dots, x_n)
    \]
    solange die Summe wohldefiniert ist, wobei wir hier über alle \( x_1 \in W_1, \dots, x_n \in W_n \) summieren.
\end{mainbox}

\begin{mainbox}{}
    Seien \( X_1, \dots, X_n \) diskrete ZV mit gemeinsamer Verteilung \( \{p(x_1, \dots, x_n)\}_{x_1 \in W_1, \; \dots, \; x_n \in W_n} \). Die folgenden Aussagen sind äquivalent
    \begin{enumerate}
        \item \( X_1, \dots, X_n \) sind unabhängig
        \item für alle \( x_1 \in W_1, \dots, x_n \in W_n \) gilt:
        \[
            p(x_1, \dots, x_n) = \P[X_1 = x_1] \cdot \dots \cdot \P[X_n = x_n]
        \]
        % d.h. die gemeinsame Gewichtsfunktion ist das Produkt der einzelnen Gewichtsfunktionen der Randverteilungen.
    \end{enumerate}
\end{mainbox}

\subsection{Gemeinsame stetige Verteilung}

\begin{mainbox}{Gemeinsame stetige Verteilung}
    Sei \( n \geq 1 \). \( X_1, \dots, X_n: \: \Omega \to \R \) besitzen eine \emph{stetige gemeinsame Verteilung}, falls \( f: \: \R^n \to \R_+ \) existiert, sodass für alle \( a_1, \dots, a_n \in \R \) gilt:
    \begin{multline*}
        \P[X_1 \leq a_1, \; \dots, \; X_n \leq a_n] \\
        = \int_{-\infty}^{a_1} \dotsi \int_{-\infty}^{a_n} f(x_1, \dots, x_n) \, \mathrm{d}x_n \, \dotsi \, \mathrm{d}x_1
    \end{multline*}
    Die Funktion \( f \) heisst die \emph{gemeinsame Dichte von \( X_1, \dots, X_n \)}.
\end{mainbox}

\begin{mainbox}{}
    Sei \( f \) die gemeinsame Dichte der ZV \( X_1, \dots, X_n \). Dann gilt
    \[
        \int_{-\infty}^\infty \dotsi \int_{-\infty}^\infty f(x_1, \dots, x_n) \, \mathrm{d}x_n \, \dotsi \mathrm{d}x_1 = 1
    \]
\end{mainbox}
Umgekehrt kann jeder Funktion \( f: \R^n \to \R_+ \), die die obige Gleichung erfüllt, ein W'raum \( (\Omega, \F, \P) \) und \( n \) ZV \( X_1, \dots, X_n \) zugeordnet werden, sodass \(  f \) die gemeinsame Dichte von \( X_1, \dots, X_n \) ist.

\paragraph{Interpretation} Betrachten wir z.B. zwei ZV \( X, Y \). Intuitiv beschreibt \( f(x, y) \, \mathrm{d}x \, \mathrm{d}y \) dabei die Wahrscheinlichkeit, dass ein Zufallspunkt \( (X, Y) \) in einem Rechteck \( [x, x + \mathrm{d}x] \times [y, y + \mathrm{d}y] \) liegt.

\begin{mainbox}{Erwartungswert des Bildes}
    Sei \( \varphi: \: \R^n \to \R \) eine Abbildung. Falls \( X_1, \dots, X_n \) eine gemeinsame Dichte \( f \) besitzen, dann lässt sich der Erwartungswert der ZV \( \varphi(X_1, \dots, X_n) \) berechnen als
    \begin{multline*}
        \E[(\varphi(X_1, \dots, X_n)] = \\
        \int_{-\infty}^{\infty} \dotsi \int_{-\infty}^\infty \varphi(x_1, \dots, x_n) f(x_1, \dots, x_n) \, \mathrm{d}x_n \, \dotsi \, \mathrm{d}x_1
    \end{multline*}
\end{mainbox}

\begin{mainbox}{Randverteilung}
    Haben \( X \) und \( Y \) die gemeinsame Verteilungsfunktion \( F \), so ist \( F_X: \: \R \to [0, 1] \) gegeben durch
    \[
        x \mapsto F_X(x) = \P[X \leq x] = \P[X \leq x, \, Y < \infty] = \lim_{y \to \infty} F(x, y)
    \]
    die Verteilungsfunktion der \emph{Randverteilung} von \( X \). Analog ist \( F_Y: \: \R \to [0, 1] \)
    \[
        y \mapsto F_Y(y) = \P[Y \leq y] = \P[X < \infty, \, Y \leq y] = \lim_{x \to \infty} F(x, y)
    \]
    die Verteilungsfunktion der Randverteilung von \( Y \).
\end{mainbox}

Haben \( X \) und \( Y \) eine gemeinsame Dichte \( f(x, y) \), so haben auch die Randverteilungen von \( X \) und \( Y \) Dichten \( f_X: \: \R \to \R_+ \) bzw. \( f_Y = \R \to \R_+ \). Analog zu den Gewichtsfunktionen \( p_X \) und \( p_Y \) gilt:
\[
    f_X(x) = \int_{-\infty}^\infty f(x, y) \, \mathrm{d}y \quad \text{und} \quad f_Y(y) = \int_{-\infty}^\infty f(x, y) \, \mathrm{d}x
\]

\begin{mainbox}{Unabhängigkeit von stetigen ZV}
    Seien \( X_1, \dots, X_n \) ZV mit Dichten \( f_{X_1}, \dots, f_{X_n} \). Dann sind die folgenden Aussagen äquivalent:
    \begin{itemize}
        \item \( X_1, \dots, X_n \) sind unabhängig
        \item \( X_1, \dots, X_n \) sind gemeinsam stetig mit gemeinsamer Dichte \( f: \: \R^n \to \R_+ \)
        \[
            f(x_1, \dots, x_n) = f_{X_1}(x_1) \cdot \dots \cdot f_{X_n}(x_n)
        \]
        % d.h. die gemeinsame Dichtefunktion \( f \) ist das Produkt der einzelnen Randdichten \( f_{X_k} \).
    \end{itemize}
\end{mainbox}

\section{Grenzwertsätze}

% \paragraph{Grundidee} In vielen Situationen taucht die Summe von vielen \emph{gleichartigen ZV} auf.
% Wir möchten wissen, wie sich diese Summe verhält, und untersuchen deshalb ihre \emph{Asymptotik}, wenn die Anzahl der Summanden gegen unendlich geht.

\paragraph{Terminologie} Die ZV \( \Bar{X}_n = \frac{1}{n} S_n = \frac{1}{n} \sum_{k = 1}^n X_k \) ist das \emph{arithmetische Mittel} der \( X_k \) und wird im Zusammenhang mit ZV auch \emph{Stichprobenmittel} genannt. Eine Realisierung des Stichprobenmittels \( \Bar{X}_n(\omega) \) wird auch \emph{empirischer Mittelwert} genannt.

\begin{itemize}
    \item \emph{Asymptotik von ZV \( \Bar{X}_n \)}: \emph{Gesetz der grossen Zahlen}
    \item \emph{Asymptotik von Verteilungen oder Verteilungsfunktionen der \( S_n \)}: \emph{zentraler Grenzwertsatz}
\end{itemize}

\subsection{Das Gesetz der grossen Zahlen}

\begin{mainbox}{Schwaches Gesetz der grossen Zahlen}
    Sei \( X_ 1, X_2, \dots \) eine Folge von unabhängigen ZV mit gleichen Erwartungswerten \( \E[X_k] = \mu \) und Varianzen \( \V[X_k] = \sigma^2 \). Sei
    \[
        \Bar{X}_n = \frac{1}{n} S_n = \frac{1}{n} \sum_{k = 1}^n X_k
    \]
    Dann \emph{konvergiert \( \Bar{X}_n \) für \( n \to \infty \) in Wahrscheinlichkeit gegen \( \mu = \E[X_k] \)}, d.h. für jedes \( \varepsilon > 0 \) gilt:
    \[
        \P[|\Bar{X}_n - \mu | > \varepsilon] \xrightarrow{n \to \infty} 0
    \]
\end{mainbox}

\subparagraph{Interpretation} Anschaulich bedeutet dies, dass mit beliebig grosser Wahrscheinlichkeit \( 1 - \delta \) der Wert von \( \Bar{X}_n \) für \( n \geq n_0 \) beliebig nahe (bis auf \( \varepsilon \)) bei \( \mu \) liegt. Dabei hängt \( n_0 \) natürlich von \( \varepsilon \) und \( \delta \) ab, nicht aber von \( \omega \).

\begin{mainbox}{Starkes Gesetz der grossen Zahlen}
    Sei \( X_1, X_2, \dots \) eine Folge von unabhängigen ZV, die alle dieselbe Verteilung haben mit endlichem Erwartungswert \( \E[X_k] \). Für
    \[
        \Bar{X}_n = \frac{1}{n} S_n = \frac{1}{n} \sum_{k = 1}^n X_k
    \]
    gilt dann
    \[
        \Bar{X}_n \xrightarrow{n \to \infty} \mu \ \text{\( \P \)-fast sicher }
    \]
    das bedeutet
    \[
        \P\left[\{\omega \in \Omega \: | \: \Bar{X}_n(\omega) \xrightarrow{n \to \infty} \mu\}\right] = 1
    \]
\end{mainbox}

\subparagraph{Interpretation} Explizit bedeutet die Aussage folgendes:
\begin{itemize}
    \item Wir geben ein \( \varepsilon > 0 \) vor,
    \item dann ist für jedes \( \omega \) ausserhalb einer Menge mit Wahrscheinlichkeit 0 die Abweichung von \( \Bar{X}_n(\omega) \) von \( \mu \) kleiner als \( \varepsilon \) für \( n \geq n_0 \),
    \item wobei nun \( n_0 \) sowohl von \( \varepsilon \) als auch von \( \omega \) abhängt.
\end{itemize}

\subsection{Zentraler Grenzwertsatz}

\begin{mainbox}{Konvergenz in Verteilung} \label{box:konvergenz_in_verteilung}
    Seien \( (X_n)_{n \in \N} \) und \( X \) ZV mit Verteilungsfunktionen \( (F_n)_{n \in \N} \) und \( F \). Wir sagen \( (X_n)_{n \in \N} \) \emph{konvergiert in Verteilung gegen \( X \)} und schreiben:
    \[
        X_n \xrightarrow{d} X \quad \text{für \( n \to \infty\)}
    \]
    falls für jeden Stetigkeitspunkt \( x \in \R \) von \( F \) gilt
    \[
        \lim_{n \to \infty} F_n(x) = \lim_{n \to \infty} \P[X_n \leq x] = \P[X \leq x] = F(x)
    \]
\end{mainbox}

\begin{mainbox}{Zentraler Grenzwertsatz}
    Sei \( (X_k)_{k \in N} \) eine Folge von i.i.d. ZV mit \( \E[X_k] = \mu \) und \( \V[X_k] = \sigma^2 \). Für die Partialsummen \( S_n = \sum_{k = 1}^n X_k \) gilt dann für alle \( x \in R \)
    \[
        \lim_{n \to \infty} \P\left[ \frac{S_n - n \mu}{\sigma \sqrt{n}} \leq x \right] = \Phi(x)
    \]
    % wobei \( \Phi \) die Verteilungsfunktion der \( \normal(0, 1) \)-Verteilung ist.
\end{mainbox}

\( S_n \) hat Erwartungswert \( \E[S_n] = n \mu \) und Varianz \( \V[S_n] = n \sigma^2 \). Also hat die Grösse
\[
    S_n^* = \frac{S_n - \mu}{\sigma \sqrt{n}} = \frac{S_n - \E[S_n]}{\sqrt{\V[S_n]}}
\]
Erwartungswert \( 0 \) und Varianz \( 1 \). Sie heisst deshalb auch die \emph{Standardisierung von \( S_n \)}.

Für praktische Anwendungen schreibt man die Aussage des ZGS meistens in der Form
\[
    \P[S_n^* \leq x] \approx \Phi(x) \quad \text{für grosse \( n \)}
\]
oder
\[
    S_n^* \stackrel{\text{approx.}}{\sim} \normal(0, 1) \quad \text{für grosse \( n \)}
\]
Zurückübersetzt für \( S_n \) und \( \Bar{X}_n = \frac{1}{n} S_n \) heisst das dann
\[
    S_n \stackrel{\text{approx.}}{\sim} \normal(n \mu, n \sigma^2) \quad \text{bzw.} \quad \Bar{X}_n \stackrel{\text{approx.}}{\sim} \normal(\mu, \frac{1}{n} \sigma^2)
\]

\paragraph{Kontinuitätskorrektur} Für \( S_n \sim \Bin(n, p) \) ist \( S_n \stackrel{\text{approx.}}{\sim} \normal(np, np(1 - p)) \) und
\begin{align*}
    \P[a < S_n \leq b] &= \P\left[ \frac{a - np}{\sqrt{np(1 - p)}} < S_n^* \leq \frac{b - np}{\sqrt{np(1 - p)}} \right] \\
    &\approx \Phi\left( \frac{b + \frac{1}{2} - np}{\sqrt{np(1 - p)}} \right) - \Phi\left( \frac{a + \frac{1}{2} - np}{\sqrt{np(1 - p)}} \right)
\end{align*}
% Die Korrektur um \( + \frac{1}{2} \) kommt dadurch, dass die Approximation der Binomial-Wahrscheinlichkeiten besser wird, wenn man die Stäbe aus dem Histogramm jeweils \emph{zentriert} über die relevanten Werte von \( a \) bis \( b \) setzt.

\subsubsection{Grosse Abweichungen und Chernoff-Schranken}

% Seien \( X_1, \dots, X_n \) i.i.d. ZV und \( S_n = \sum_{k = 1}^n X_k \)
% \begin{itemize}
%     \item Wir wollen die Wahrscheinlichkeit dafür abschätzen, dass \( S_n \) sehr gross wird, indem wir obere Schranken für \( \P[S_n \geq b] \) angeben, wobei typisch \( b \) gross ist.
%     \item Analog kann man natürlich auch obere Schranken für \( \P[S_n \leq -a] \) mit \( a > 0 \) gross angeben.
% \end{itemize}

\begin{mainbox}{Momenterzeugende Funktion}
    Die \emph{momenterzeugende Funktion} von \( X \) ist für \( t \in \R \) definiert durch
    \[
        M_X(t) = \E[e^{tX}]
    \]
    Das ist immer wohldefiniert in \( [0, \infty] \), kann aber \( +\infty \) werden.
\end{mainbox}

% \paragraph{Bernoulli-Verteilung \( X \sim \Ber(p) \)}
% \[
%     M_X(t) = \E[e^{tX}] = e^t \cdot p + 1 \cdot (1 - p) = 1- p + pe^t
% \]

% \paragraph{Binomialverteilung \( X \sim \Bin(n, p) \)}
% \begin{multline*}
%     M_X(t) = \E[e^{tX}] = \sum_{k = 0}^n e^{tk} \binom{n}{k} p^k(1 - p)^{n - k} \\
%     = \sum_{k = 0}^n \binom{n}{k} (e^tp)^k(1 - p)^{n - k} = (1 - p + pe^t)^n
% \end{multline*}

\begin{mainbox}{Chernoff-Ungleichung}
    Seien \( X_1, \dots, X_n\) i.i.d., wessen \( M_X(t) \) für alle \( t \in \R \) endlich ist. Für jedes \( b \in \R \) gilt dann:
    \[
        \P[S_n \geq b] \leq \exp\left( \inf_{t \in \R}(n \log M_X(t) - tb) \right)
    \]
\end{mainbox}

\begin{mainbox}{Chernoff-Schranke}
    Seien \( X_1, \dots, X_n \) unabhängig mit \( X_k \sim \Ber(p_k) \) und sei \( S_n = \sum_{k = 1}^n X_k \). Sei \( \mu_n = \E[S_n] = \sum_{k = 1}^n p_k \) und \( \delta > 0 \), dann gilt:
    \[
        \P[S_n \geq (1 + \delta) \mu_n] \leq \left( \frac{e^\delta}{(1 + \delta)^{1 + \delta}} \right)^{\mu_n}
    \]
\end{mainbox}

\section{Schätzer}

\subsection{Statistische Grundideen}

% Wir erhalten Daten \( x_1, \dots, x_n \) und fassen diese als \emph{Realisierungen} \( X_1(\omega), \dots, X_n(\omega) \) von ZV \( X_1, \dots, X_n \) auf.
% Unter geeigenten Zusatzannahmen suchen wir nach Aussagen über die Verteilung von \( X_1, \dots, X_n \).

Die Gesamtheit der Beobachtungen \( x_1, \dots, x_n \) oder ZV \( X_1, \dots, X_n \) nennt man oft eine \emph{Stichprobe}. Die Anzahl \( n \) heisst dann der \emph{Stichprobenumfang}.

\paragraph{Parametrische statistische Analyse}
\begin{enumerate}
    \item \emph{Deskriptive Statistik}
    \item \emph{Wahl eines (parametrischen) Modells}: Spezifiziere eine Parametermenge und Familie \( (\P_\vartheta)_{\vartheta \in \Theta} \) von Modellen.
    \item \emph{Schätzung der Parameter}: \emph{Schätzer} benutzen. Die zugehörige \emph{Schätzfunktion} ordnet gegebenen Daten \( x_1, \dots, x_n \) einen Parameter \( \vartheta \in \Theta \) zu.
    \item \emph{Kritische Modellüberprüfung (Anpassungstest)}: Prüfen, ob die Daten zum gewählten \( \vartheta \) bzw. \( \P_\vartheta \) gut passen mit geeignetem \emph{statistischen Test}.
    % \item \emph{Aussagen über Zuverlässigkeit der Schätzungen}: Man kann auch versuchen, einen Bereich in \( \Theta \) so zu spezifizieren, dass alle zugehörigen Modelle \( \P_\vartheta \) in einem zu präzisierenden Sinn gut zu den Daten passen. Man spricht dann von einem \emph{Konfidenzbereich}.
\end{enumerate}

\subsection{Schätzer}

Sei \( X_1, \dots, X_n \) eine Stichprobe, für die wir ein Modell suchen.
\begin{itemize}
    \item Wir haben \( \Theta \) und für jedes \( \vartheta \in \Theta \) ein \( (\Omega, \F, \P_\vartheta) \).
    \item Meistens ist \( \Theta \in \R^m \), und wir suchen dann für \( \vartheta_1, \dots, \vartheta_m \) \emph{Schätzer} \( T_1, \dots, T_m \) aufgrund unserer Stichprobe.
\end{itemize}

\begin{mainbox}{Schätzer}
    Ein \emph{Schätzer} ist eine ZV der Form
    \[
        T_l = t_l(X_1, \dots, X_n)
    \]
    Die \emph{Schätzfunktionen} \( t_l: \: \R^n \to \R \) müssen noch gewählt / gefunden werden.
    Einsetzen von Daten \( x_k = X_k(\omega), \ k = 1, \dots, n \) liefert dann \emph{Schätzwerte} \( T_l(\omega) = t_l(x_1, \dots, x_n) \) für \( \vartheta_l, \ l = 1, \dots, m \).
    Der Kürze halber schreiben wir oft auch \( T = (T_1, \dots, T_m) \) und \( \vartheta = (\vartheta_1, \dots, \vartheta_m) \).
\end{mainbox}

% \subparagraph{Wichtig!} Die Konzepte \say{Schätzer} und \say{Schätzwert} müssen sauber auseinandergehalten werden.
% \begin{itemize}
%     \item Ein Schätzwert ist eine Zahl
%     \[
%         T_l(\omega) = t_l(X_1(\omega), \dots, X_n(\omega)) = t_l(x_1, \dots, x_n) \in \R \text{,}
%     \]
%     also die Realisierung der ZV \( T_l \) im Experiment \( \omega \)
%     \item Ein Schätzer \( T_l \) ist eine ZV. Er hat also eine Verteilung unter \( \P_\vartheta \) für jedes \( \vartheta \in \Theta \). Wir können also z.B. nach der Wahrscheinlichkeit (im Modell \( P_\vartheta \)) fragen, mit der er nahe beim wahren, uns unbekannten Parameter \( \vartheta \) liegt.
% \end{itemize}
% Also ist der Schätzer eine \emph{Funktion oder Vorschrift}, die uns die Berechnungsmethode angibt, und der Schätzwert ist das \emph{Ergebnis einer konkreten Berechnung}.

\begin{mainbox}{Erwartungstreue}
    Ein Schätzer \( T \) ist \emph{erwartungstreu} für \( \vartheta \), falls für alle \( \vartheta \in \Theta \) gilt:
    \[
        \E_\vartheta[T] = \vartheta \text{.}
    \]
\end{mainbox}
\subparagraph{Interpretation} Im Mittel (über alle denkbaren Realisierungen \( \omega \)) schätzt \( T \) richtig, und zwar unabhängig davon, welches \( P_\vartheta \) zu Grunde liegt.

\begin{mainbox}{Bias und MSE}
    Sei \( \vartheta \in \Theta \) und \( T \) ein Schätzer.
    \begin{itemize}
        \item Der \emph{Bias} (oder \emph{erwartete Schätzfehler}) von \( T \) in \( P_\vartheta \) ist:
        \[
            \E_\theta[T] - \vartheta \text{.}
        \]
        Es gilt also: \( \E_\theta[T] - \vartheta = 0 \) \( \iff \) \( T \) ist erwartungstreu
        \item Der \emph{mittlere quadratische Schätzfehler (MSE)} von \( T \) in \( \P_\vartheta \) ist:
        \[
            \mse_\vartheta[T] = \E_\vartheta[(T - \vartheta)^2] \text{.}
        \]
    \end{itemize}
\end{mainbox}

\begin{mainbox}{}
    Eine \emph{Folge von Schätzern} \( T^{(n)}, \ n \in \N \) heisst \emph{konsistent} für \( \vartheta \), falls \( T^{(n)} \) für \( n \to \infty \) in \( \P_\theta \)-Wahrscheinlichkeit gegen \( \vartheta \) konvergiert, d.h. für jedes \( \vartheta \in \Theta \) und jedes \( \varepsilon > 0 \) gilt
    \[
        \lim_{n \to \infty} \P_\vartheta[|T^{(n)} - \vartheta| > \varepsilon] = 0 \text{.}
    \]
    % (Das setzt anschaulich voraus, dass man beliebig viele Daten haben könnte.)
\end{mainbox}

Man kann den MSE zerlegen als
\[
    \mse_\vartheta[T] = \E_\vartheta[(T - \vartheta)^2] = \V_\theta[T] + (\E_\theta[T] - \theta)^2 \text{,}
\]
Für erwartungstreue Schätzer sind somit Varianz und MSE dasselbe.

\subsection{ML-Methode}

Wir schreiben oft kurz \( \Vec{X} = (X_1, \dots, X_n) \).
In jedem \( \P_\vartheta \) sind \( X_1, \dots, X_n \) entweder
\begin{itemize}
    \item \emph{diskret mit gemeinsamer Gewichtsfunktion} \( p_{\Vec{X}}(x_1, \dots, x_n; \vartheta) \), oder
    \item \emph{stetig mit gemeinsamer Dichtefunktion} \( f_{\Vec{X}}(x_1, \dots, x_n; \vartheta) \).
\end{itemize}

Meistens sind \( X_k \) unter \( \P_\vartheta \) sogar i.i.d. mit individueller Gewichtsfunktion \( p_X(x;\vartheta) \) bzw. Dichtefunktion \( f_X(x; \vartheta) \).
Dann ist die gemeinsame Gewichtsfunktion
\[
    p_{\Vec{X}}(x_1, \dots, x_n; \vartheta) = \prod_{k = 1}^n p_X(x_k; \vartheta)
\]
bzw. die gemeinsame Dichtefunktion
\[
    f_{\Vec{X}}(x_1, \dots, x_n; \vartheta) = \prod_{k = 1}^n f_X(x_k; \vartheta) \text{.}
\]
% Anschaulich ist
% \[
%     p_{\Vec{X}}(x_1, \dots, x_n; \vartheta) = \P_\vartheta[X_1 = x_1, \dots, X_n = x_n]
% \]
% gerade die Wahrscheinlichkeit in \( \P_\vartheta \), dass \( X_1, \dots, X_n \) die Werte \( x_1, \dots, x_n \) liefert, und \( f_{\Vec{X}}(x_1, \dots, x_n; \vartheta) \) ist das stetige Analogon.

\begin{mainbox}{Likelihood-Funktion}
    Die \emph{Likelihood-Funktion} ist
    \[
        L(x_1, \dots, x_n; \vartheta) = \begin{cases}
            p_{\Vec{X}}(x_1, \dots, x_n; \vartheta) & \text{im diskreten Fall} \\
            f_{\Vec{X}}(x_1, \dots, x_n; \vartheta) & \text{im stetigen Fall}
        \end{cases}
    \]
    \( \log L(x_1, \dots, x_n; \vartheta) \) heisst \emph{\( \log \)-Likelihood-Funktion} und hat im i.i.d.-Fall den Vorteil, durch eine Summe gegeben zu sein.
\end{mainbox}

\begin{itemize}
    \item Für \( X_1, \dots, X_n \) gibt uns \( L(x_1, \dots, x_n; \vartheta) \), zumeindest im diskrete Fall, die Wahrscheinlichkeit in \( \P_\vartheta \), dass unsere Stichprobe die Werte \( x_1, \dots, x_n \) liefert.
    \item Um eine möglichst gute Anpassung des Modells an die Daten zu erreichten, wollen wir diese Wahrscheinlichkeit maximieren, indem wir den Paramenter \say{geschickt} wählen.
\end{itemize}

\begin{mainbox}{ML-Schätzer}
    Der \emph{ML-Schätzer} \( T_\text{ML} \) für \( \vartheta \) maximiert die Funktion
    \[
        \vartheta \mapsto L(X_1, \dots X_n; \vartheta)
    \]
    über alle \( \vartheta \), d.h.
    \[
        T_\text{ML} = t_\text{TM}(X_1, \dots, X_n) \in \mathop{\arg \max}_{\vartheta \in \Theta} \ L(X_1, \dots, X_n; \vartheta)
    \]
\end{mainbox}
Meistens sind \( X_1, \dots, X_n \) i.i.d. unter \( \P_\vartheta \). Dann ist \( L \) ein Produkt, und es ist bequemer stattdessen \( \log L \) zu maximieren, weil dies eine Summe ist.
Statt zu maximieren sucht man ferner meinstens nur Nullstellen der Ableitung nach \( \vartheta \).

\subparagraph{Bemerkung} In Rechnungen arbeitet man oft mit \( L(x_1, \dots, x_n; \vartheta) \), insbesondere beim Maximieren über \( \vartheta \).
\begin{itemize}
    \item Das optimale \( \vartheta^* \) ist dann eine Funktion \( t_\text{ML} \) von \( x_1, \dots, x_n \).
    \item Damit der resultierende Schätzer \( T_\text{ML} \) von der Stichprobe \( X_1, \dots, X_n \) abhängt, muss \( x_1, \dots, x_n \) durch \( X_1, \dots, X_n \) ersetzt werden.
    \item Also ist der ML-Schätzer \( T_\text{ML} = t_\text{ML}(X_1, \dots, X_n) \).
\end{itemize}

% \paragraph{Bernoulli-Verteilung (\(  X_1, \dots, X_n \ \text{i.i.d.} \sim \Ber({p}) \))} Der ML-Schätzer für \( \vartheta = p \) ist
% \[
%     T = \frac{1}{n} \sum_{k = 1}^n X_k = \Bar{X}_n
% \]

% \paragraph{Normalverteilung (\( X_1, \dots, X_n \ \text{i.i.d.} \sim \normal(\mu, \sigma^2) \))} Der ML-Schätzer für \( \vartheta = (\mu, \sigma^2) \) ist
% \begin{align*}
%     T_1 &= \frac{1}{n} \sum_{k = 1}^n X_k = \Bar{X}_n \\
%     T_2 &= \frac{1}{n} \sum_{k = 1}^n (X_k - \Bar{X}_n)^2 = \frac{1}{n} \sum_{k = 1}^n X_k^2 - (\Bar{X}_n)^2
% \end{align*}

Der Schätzer \( T = (T_1, T_2) \) für die Normalverteilung ist ganz allgemein, in jedem \( \P_\vartheta \), in dem \( X_1, \dots, X_n \) i.i.d. sind, der sogenannte \emph{Momentenschätzer} für
\[
    (\E_\vartheta[X], \V_\vartheta[X]) \text{.}
\]
Er ist aber \emph{nicht erwartungstreu} für \( (\E_\vartheta[X], \V_\vartheta[X]) \).
Um einen erwartungstreuen Schätzer \( T' \) für \( (\E_\vartheta[X], \V_\vartheta[X]) \) zu erhalten, benutzt man deshalb meistens
\begin{align*}
    T'_1 &= T_1 = \Bar{X}_n, \quad \text{das \emph{Stichprobenmittel} und} \\
    T'_2 &= \frac{n}{n - 1} T_2 = \frac{1}{n - 1} \sum_{k = 1}^n X_k^2 - \frac{n}{n - 1} (\Bar{X}_n)^2
\end{align*}
Für \( T'_2 \) benutzt man oft die Notation
\[
    S^2 = \frac{1}{n - 1} \sum_{k = 1}^n (X_k - \Bar{X}_n)^2
\]
Man nennt die Schätzer für \( \V_\vartheta[X] \) \emph{(korrigierte) empirische (Stichproben-)varianz}.

\subsection{Verteilungsaussagen}

\subsubsection{Approx. Zugang über ZGS}

% In vielen Situationen ist es nützlich oder nötig, die Verteilung (unter \( \P_\vartheta \) für jedes \( \vartheta \in \Theta \) oder für gewisse \( \vartheta \)) eines Schätzers zu kennen. \emph{Exakte allgemeine Aussagen} gibt es dazu nicht viele.
Einen allgememinen \emph{approximativen Zugang} liefert der zentrale Grenzwertsatz:
\begin{itemize}
    \item Oft ist ein \( T \) eine Funktion einer Summe \( \sum_{k = 1}^n Y_k \), wobei die \( Y_k \) in \( \P_\vartheta \) i.i.d. sind.
    \item Das einfachste Beispiel ist \( T = \Bar{Y}_n = \frac{1}{n} \sum_{k = 1}^n Y_k \).
    \item Nach dem zentralen Grenzwertsatz ist dann für grosse \( n \)
    \[
        \sum_{k = 1}^n Y_k \text{ approximativ normalverteilt unter \( \P_\vartheta \)}
    \]
    mit Parametern \( \mu = n \E_\vartheta[Y_k] \) und \( \sigma^2 = n \V_\vartheta[Y_k] \).
    % \item Das kann man benutzen, um für die Verteilung von \( T \) approximative Aussagen zu bekommen und damit gewisse Fragen zumindest approximativ zu beantworten.
\end{itemize}

\subsubsection{\( \chi^2 \)-Verteilung}

\begin{mainbox}{\( \chi^2 \)-Verteilung}
    Eine stetige ZV \( X \) heisst \emph{\( \chi^2 \)-verteilt mit \( m \) Freiheitsgraden} (\( X \sim \chi^2_m \)) falls ihre Dichte für \( x \geq 0 \) gegeben ist durch:
    \[
        f_X(x) = \frac{1}{2^\frac{m}{2} \Gamma\left(\frac{m}{2}\right)} x^{\frac{m}{2} - 1} e^{-\frac{x}{2}}
    \]
\end{mainbox}

\begin{subbox}{Gammafunktion}
    \begin{itemize}
        \item \( \Gamma \) heisst \emph{(Eulersche) Gammafunktion} und ist für \( x \geq 0 \):
        \[
            \Gamma(x) = \int_0^\infty t^{x - 1} e^{-t} \, \mathrm{d}t
        \]
        \item Es gilt:
        \[
            \Gamma(n + 1) = n! \quad \text{für } n \in \N_0
        \]
    \end{itemize}
\end{subbox}

\begin{itemize}
    \item \emph{Entstehung der \( \chi^2_m \)-Verteilung}: Sind die ZV \( X_1, \dots, X_m \ \text{i.i.d.} \sim \normal(0, 1) \), so ist die Summe ihrer Quadrate \( \chi^2_m \)-verteilt
    \[
        \sum_{k = 1}^n X_k^2 \sim \chi^2_m
    \]
    \item Die \( \chi^2_m \)-Verteilung ist ein \emph{Spezialfall einer \( \mathop{Ga} \)-Verteilung} mit \( \alpha = \frac{m}{2} \) und \( \lambda = \frac{1}{n} \).
    \item Mit \( m = 2 \) erhält man eine \emph{Exponentialverteilung} mit Parameter \( \frac{1}{2} \).
\end{itemize}

\subsubsection{Studentsche \( t \)-Verteilung}

\begin{mainbox}{Studentsche \( t \)-Verteilung}
    Eine stetige ZV \( X \) heisst \emph{\( t \)-verteilt mit \( m \) Freiheitsgraden} (\( X \sim t_m \)) falls ihre Dichte für \( x \in \R \) gegeben ist durch:
    \[
        f_X(x) = \frac{\Gamma\left(\frac{m + 1}{2}\right)}{\sqrt{m \pi} \Gamma\left(\frac{m}{2}\right)} \left(1 + \frac{x^2}{m}\right)^{-\frac{m + 1}{2}}
    \]
\end{mainbox}

\begin{itemize}
    \item \emph{Entstehung der \( t \)-Verteilung}: Sind \( X \sim \normal(0, 1) \) und \( Y \sim \chi^2_m \) unabhängig, so ist der folgende Quotient \( t \)-verteilt:
    \[
        \frac{X}{\sqrt{\frac{1}{m} Y}} \sim t_m
    \]
    \item Für \( m = 1 \) ist das eine Cauchy-Verteilung.
    \item Für \( m \to \infty \) erhält man asymptotisch eine \( \normal(0, 1) \)-Verteilung.
    \item Sie ist symmetrisch um \( 0 \) und \emph{langschwänziger} als die Normalverteilung.
\end{itemize}

\subsubsection{Normalverteilung}

\begin{subbox}{}
    Seien \( X_1, \dots, X_n \sim \normal(\mu, \sigma^2) \) i.i.d. und
    \[
        \begin{array}{cc}
            \Bar{X}_n = \sum_{k = 1}^n X_k, & S^2 = \frac{1}{n - 1} \sum_{k = 1}^n (X_k - \Bar{X}_n)^2 \text{.}
        \end{array}
    \]
    Es gelten folgende Aussagen:
    \begin{enumerate}
        \item \( \Bar{X}_n \sim \normal(\mu, \frac{1}{n} \sigma^2) \), also \( \frac{\Bar{X}_n - \mu}{\frac{\sigma}{\sqrt{n}}} \sim \normal(0, 1) \).
        \item \( \frac{n - 1}{\sigma^2} S^2 = \frac{1}{\sigma^2} \sum_{k = 1}^n (X_k - \Bar{X}_n)^2 \sim \chi^2_{n - 1} \).
        \item \( \Bar{X}_n \) und \( S^2 \) sind unabhängig.
        \item \( \frac{\Bar{X}_n - \mu}{\frac{S}{\sqrt{n}}} = \frac{\frac{\Bar{X}_n - \mu}{\frac{\sigma}{\sqrt{n}}}}{\frac{S}{\sigma}} = \frac{\frac{\Bar{X}_n - \mu}{\frac{\sigma}{\sqrt{n}}}}{\sqrt{\frac{1}{n - 1} \frac{n - 1}{\sigma^2} S^2}} \sim t_{n - 1} \)
    \end{enumerate}
\end{subbox}

\section{Tests}

\subsection{Grundbegriffe}

% Nun haben wir schon eine Vermutung, wo in \( \Theta \) der (unbekannte) richtige Parameter \( \vartheta \) liegen könnte, und wollen diese mit Hilfe der Daten überprüfen (\say{testen}).

\subsubsection{Hypothesen}

Für eine \emph{Hypothese \( \Theta_0 \subseteq \Theta \)} und eine \emph{Alternative \( \Theta_A \subseteq \Theta \)} ist \( \Theta_0 \cap \Theta_A = \emptyset \).
Die Hypothese wird auch \emph{Nullhypothese} und die Alternative wird auch \emph{Alternativhypothese oder Gegenhypothese} genannt.
Meist schreibt man das als
\begin{gather*}
    \text{Hypothese } H_0: \: \vartheta \in \Theta_0 \text{,} \\
    \text{Alternative } H_A: \: \vartheta \in \Theta_A \text{.}
\end{gather*}
Ist keine explizite Alternative spezifiziert, so hat man
\[
    \Theta_A = \Theta_0^\complement = \Theta \setminus \Theta_0 \text{.}
\]
Eine Hypothese und / oder Alternative heissen
\begin{itemize}
    \item \emph{einfach}, falls \( \Theta_0 = \{\vartheta_0\} \) bzw. \( \Theta_A = \{\vartheta_A\} \),
    \item sonst heissen sie \emph{zusammengesetzt}.
\end{itemize}

% Explizierter formuliert ist also die Hypothese
% \[
%     H_0: \: \text{\say{der wahre (unbekannte) Parameter \( \vartheta \) liegt in der Menge \( \Theta_0 \)}}
% \]
% und die Alternative
% \[
%     H_A: \: \text{\say{der wahre Parameter liegt in \( \Theta_A \).}}
% \]

\subsubsection{Teststatistik und Verwerfungsbereich}

\begin{mainbox}{Test}
    Ein \emph{Test} ist ein Paar \( (T, K) \), wobei
    \begin{itemize}
        \item die \emph{Teststatistik \( T \)} eine ZV der Form \( T = t(X_1, \dots, X_n) \) ist, mit einer messbaren Funktion \( t: \: \R^n \to \R \), und
        \item der \emph{kritische Bereich} oder \emph{Verwerfungsbereich \( K \subseteq \R \)} eine (deterministische) Teilmenge von \( \R \) ist.
    \end{itemize}
\end{mainbox}

\paragraph{Entscheindungsregel}
\begin{itemize}
    \item Die Hypothese \( H_0 \) wird verworfen, wenn \( T(\omega) \in K \).
    \item Die Hypothese \( H_0 \) wird angenommen, wenn \( T(\omega) \notin K \).
\end{itemize}

% Die Entscheidung des Tests \emph{hängt von \( \omega \) ab}. Weil \( T \) eine ZV ist, ist die Menge
% \[
%     \{T \in K\} = \{\omega \in \Omega \: | \: T(\omega) \in K\}
% \]
% eine (messbare) Teilmenge von \( \Omega \), also ein Ereignis. Somit können wir ihre Wahrscheinlichkeit \( \P_\vartheta[T \in K] \) in jedem Modell \( \P_\vartheta \) betrachten.

\subsubsection{Fehlerarten}

\begin{itemize}
    \item \emph{Fehler 1. Art}: Die Hypothese wird abgelehnt, obwohl sie richtig ist.
    \begin{itemize}
        \item Das passiert für \( \vartheta \in \Theta_0 \) und \( T \in K \).
        \item \( \P_\vartheta[T \in K] \) heisst für \( \vartheta \in \Theta_0 \) die \emph{Wahrscheinlichkeit für einen Fehler 1. Art}.
    \end{itemize}
    \item \emph{Fehler 2. Art}: Die Hypothese wird angenommen, obwohl sie falsch ist.
    \begin{itemize}
        \item Das passiert für \( \vartheta \in \Theta_A \) und \( T \notin K \).
        \item \( \P_\vartheta[T \notin K] = 1 - \P_\vartheta[T \in K] \) heisst für \( \vartheta \in \Theta_A \) die \emph{Wahrscheinlichkeit für einen Fehler 2. Art}.
    \end{itemize}
\end{itemize}

\subsubsection{Signifikanzniveau und Macht}

\begin{enumerate}
    \item Man wählt zuerst ein \emph{Signifikanzniveau \( \alpha \in (0, 1) \)} und verlangt
    \[
        \sup_{\vartheta \in \Theta_0} \P_\vartheta[T \in K] \leq \alpha \text{.}
    \]
    Man kontrolliert also die Wahrscheinlichkeit für einen Fehler 1. Art durch \( \alpha \).
    \item Anschliessend versucht man, die \emph{Macht} des Tests
    \[
        \beta: \: \Theta_A \to [0, 1], \quad \vartheta \mapsto \beta(\vartheta) = \P_\vartheta[T \in K]
    \]
    möglichst gross zu bekommen, soweit das noch möglich ist.
    Äquivalent formuliert heisst das, dass man \( 1 - \beta(\vartheta) = \P_\vartheta[T \notin K] \) für \( \vartheta \in \Theta_A \), also die Wahrscheinlichkeit für einen Fehler 2. Art, möglichst klein bekommen will.
\end{enumerate}

\begin{itemize}
    \item Das obige asymmetrische Verfahren macht es schwieriger, die Hypothese zu verwerfen als sie beizubehalten.
    \item Ein seriöser Test wird deshalb immer die \emph{Negation der eigentlich gewünschten Aussage} als Hypothese benutzen.
    % \item Gelingt es dann nämlich, die Hypothese trotz erschwerter Bedingungen zu verwerfen, so kann man zuversichtlicher sein, tatsächlich einen Effekt gefunden zu haben.
\end{itemize}


\subsection{Konstruktion von Tests}

\subsubsection{Likelihood-Quotient}

Seien \( X_1, \dots, X_n \) ZV, die entweder diskret oder gemeinsam stetig unter \( \P_{\vartheta_0} \) und \( \P_{\vartheta_A} \) sind.
% Sei \( L(x_1, \dots, x_n; \vartheta) \) die Likelihood-Funktion.

\begin{mainbox}{Likelihood-Quotient}
    Für \( \vartheta_0 \in \Theta_0 \), \( \vartheta_A \in \Theta_A \) und \( x_1, \dots, x_n \in \R \) definieren wir den \emph{Likelihood-Quotienten} durch
    \[
        R(x_1, \dots, x_n; \vartheta_0, \vartheta_A) = \frac{L(x_1, \dots, x_n; \vartheta_A)}{L(x_1, \dots, x_n; \vartheta_0)} \text{.}
    \]
    \( R(x_1, \dots, x_n; \vartheta_0, \vartheta_A) := +\infty \), wenn der Nenner 0 ist.
\end{mainbox}

\begin{mainbox}{Likelihood-Quotienten-Test}
    Sei \( c \geq 0 \). Der \emph{Likelihood-Quotienten-Test} mit Parameter \( c \) ist ein Test \( (T, K) \) mit Teststatistik \( T = R(X_1, \dots, X_n; \vartheta_0, \vartheta_A) \) und Verwerfungsbereich \( K = (c, \infty] \).
\end{mainbox}

Sind Hypothese und Alternative beide einfach, so ist dieser Test in folgendem Sinne \emph{optimal}.

\begin{mainbox}{Neyman-Pearson-Lemma}
    Sei \( \Theta_0 = \{\vartheta_0\} \) und \( \Theta_A = \{\vartheta_A\} \). Sei \( (T, K) \) ein Likelihood-Quotienten-Test mit Parameter \( c \) und Signifikanzniveau \( \alpha^* \coloneqq \P_{\vartheta_0}[T \in K] \). Ist \( (T', K') \) ein anderer Test mit Signifikanzniveau \( \P_{\vartheta_A}[T' \in K'] \eqqcolon \alpha \leq \alpha^* \), so gilt auch
    \[
        \P_{\vartheta_A}[T' \in K'] \leq \P_{\vartheta_A}[T \in K] \text{.}
    \]
\end{mainbox}
Das bedeutet, jeder andere Test mit kleinerem Signifikanzniveau hat auch kleinere Macht.

% Die obige Situation mit einfacher Hypothese und Alternative ist so speziell, dass sie in der Praxis kaum je auftritt. Die Grundidee für den Test lässt sich aber verallgemeinern und liefert in gewissen (weniger restriktiven) Situationen immer noch gute oder optimale Tests. Dieses Vorgehen kann man also mit gutem Gewissen als einen \emph{systematischen Ansatz empfehlen}.

Bei zusammengesetzten Hypothesen und Alternativen betrachtet man den sogenannten \emph{verallgemeinerten Likelihood-Quotienten}.
\begin{mainbox}{}
    Der \emph{verallgemeinerte Likelihood-Quotient} ist gegeben durch
    \[
        R(x_1, \dots, x_n) = \frac{\sup_{\vartheta \in \Theta_A} L(x_1, \dots, x_n; \vartheta)}{\sup_{\vartheta \in \Theta_0} L(x_1, \dots, x_n; \vartheta)}
    \]
    oder auch
    \[
        \Tilde{R}(x_1, \dots, x_n) = \frac{\sup_{\vartheta \in \Theta_A \cup \Theta_0} L(x_1, \dots, x_n; \vartheta)}{\sup_{\vartheta \in \Theta_0} L(x_1, \dots, x_n; \vartheta)} \text{.}
    \]
\end{mainbox}

Nun wählt man als Teststatistik \( T_0 = R(X_1, \dots, X_n) \) bzw. \( \Tilde{T} = \Tilde{R}(X_1, \dots, X_n) \) mit kritischem Bereich \( K_0 = (c_0, \infty) \).
Durch Umformen erhält man daraus oft einen äquivalenten, aber einfacheren Test \( (T, K) \).
\( c_0 \) bzw. \( K \) muss man so wählen, dass der Test das gewählte Signifikanzniveau einhält.

\subsection{Besipiele}

\subsubsection{Gauss-Test (oder \( Z \)-Test})

Hier sind \( X_1, \dots, X_n \sim \normal(\vartheta, \sigma^2) \) i.i.d unter \( \P_\vartheta \) mit \emph{bekannter Varianz} \( \sigma^2 \).

Wir wollen die Hypothese \( H_0: \: \vartheta = \vartheta_0 \) testen.

Mögliche Alternativen \( H_A \) sind \( \vartheta > \vartheta_0 \) oder \( \vartheta < \vartheta_0 \) (einseitig) oder \( \vartheta \neq \vartheta_0 \) (zweiseitig).

Die Teststatistik ist in jedem Fall
\[
    T = \frac{\Bar{X}_n - \vartheta_0}{\frac{\sigma}{\sqrt{n}}} \sim \normal(0, 1) \quad \text{unter } \P_{\vartheta_0} \text{.}
\]

Der kritische Bereich \( K \) ist von der Form
\begin{itemize}
    \item \( (c_>, \infty) \) bzw. \( (-\infty, c_<) \) für den einseitigen Test gegen die Alternative \( H_A: \: \vartheta > \vartheta_0 \) bzw. \( H_A: \: \vartheta < \vartheta_0 \), oder
    \item \( (-\infty, -c_{\neq}) \cup (c_{\neq}, \infty) \) für den zweiseitigen Fall. Hier verwirft man \( H_0 \) also zugunsten der Alternative \( H_A: \: \vartheta \neq \vartheta_0 \), falls \( |T| > c_{\neq} \) ist.
\end{itemize}

Die Konstanten \( c_>, c_<, c_{\neq} \) bestimmt man zum gewählten Niveau mit Hilfe der Verteilung von \( T \) unter \( \P_{\vartheta_0} \).

Zum Beispiel liefert die Bedingung
\[
    \alpha = \P_{\vartheta_0}[T \in K_>] = \P_{\vartheta_0}[T > c_>] = 1 - \P_{\vartheta_0}[T \leq c_>] = 1 - \Phi(c_>) \text{,}
\]
dass \( c_> = \Phi^{-1}(1 - \alpha) \eqqcolon z_{1 - \alpha} \) das sogenannte \emph{\( (1 - \alpha) \)-Quantil} der \( \normal(0, 1) \)-Verteilung sein muss. Für \( \vartheta > \vartheta_0 \) verwirft man also \( H_0 \), falls gilt
\[
    \Bar{X}_n > \vartheta_0 + z_{1 - \alpha} \frac{\sigma}{\sqrt{n}} \text{.}
\]

Analog ist \( c_< = z_\alpha = -z_{1 - \alpha} \) und \( c_{\neq} = z_{1 - \frac{\alpha}{2}} \), wobei wir die Symmetrie der \( \normal(0, 1) \)-Verteilung ausnutzen. Es gilt nämlich
\[
    \alpha = \P_{\vartheta_0}[T < c_<] = \P_{\vartheta_0}[T > -c_<] \quad \text{für} \ -c_< = z_{1 - \alpha}
\]
und
\[
    \begin{split}
        \alpha &= \P_{\vartheta_0}[T \in K_{\neq}] = \P_{\vartheta_0}[T < -c_{\neq}] + \P_{\vartheta_0}[T > c_{\neq}] \\
        &= \Phi(-c_{\neq}) + 1 - \Phi(c_{\neq}) = 2(1 - \Phi(c_{\neq})) \text{.}
    \end{split}
\]

Die benötigten Quantile findet man in der Tabelle für \( \Phi \).
Dort kann man den Wert \( 1 - \alpha \) suchen und zurück zu \( \Phi^{-1}(1 - \alpha) = z_{1 - \alpha} \) gehen.
Alternativ findet man Quantili der Standard-Normalverteilung oft auch in Tabellen der \( t \)-Verteilung für Anzahl der Freiheitsgrade \( n = \infty \).

    \int_a^b f(x) \cdot g'(x) \, \mathrm{d}x = \big[ f(x) \cdot g(x) \big]_a^b - \int_a^b f'(x) \cdot g(x) \, \mathrm(d)x
\end{gather*}
