\section{Gewöhnliche Differentialgleichungen}

\begin{subbox}{Ordnung einer Differentialgleichung}
    Ordnung der höchsten Ableitung, die in der Differentialgleichung vorkommt
\end{subbox}

\begin{subbox}{Linearität}
    \[
        a(x)y(x) + b(x) \text{, } c(x)y'(x) + d(x) \text{, etc.}
    \]
    Es dürfen also keine nicht linearen Terme wie \( y^2 \), \( (y'')^3 \), \( \sin(y) \), \( e^{y'} \), etc. vorkommen.
\end{subbox}

\begin{subbox}{Homogenität}
    Es kommen keine Terme vor, die rein von \( x \) abhängen. Sonst nennt man die Gleichung \emph{inhomogen}.
\end{subbox}

\begin{mainbox}{Anfangswertproblem}
    Ein \emph{Anfangswertproblem} \( n \)-ter Ordnung ist eine ODE \( n \)-ter Ordnung zusammen mit \( n \) Anfangsbedingungen:
    \[
        \begin{array}{l}
            F(x, y(x), y'(x), \dots, y^{(n)}(x)) = 0 \\
            y(x_0) = y_0, \quad
            y'(x_0) = y_1, \quad
            \dots, \quad
            y^{(n - 1)}(x_0) = y_{n - 1}
        \end{array}
    \]
\end{mainbox}


\subsection{ODEs \emph{1. Ordnung}}

\subsubsection{\emph{Homogene} ODEs 1. Ordnung}

\begin{mainbox}{Separierbar}
    \[
        y'(x) = a(x) \cdot b(y)
    \]
\end{mainbox}

\paragraph{Separation der Variablen}

\begin{align*}
    y' &= a(x) \cdot b(y) \\
    \odv{y}{x} &= a(x) \cdot b(y) \\
    \frac{1}{b(y)} \odif{y} &= a(x) \odif{x} \\
    \int \frac{1}{b(y)} \odif{y} &= \int a(x) \odif{x} \\
    \int \frac{1}{b(y)} \odif{y} &= A(x) + C
\end{align*}

Falls die ODE linear ist (d.h. \( y' + ay = 0 \)) gilt
\[
    y = z \cdot \exp(-A(x))
\]


\subsubsection{\emph{Inhomogene} ODEs 1. Ordnung}

\begin{mainbox}{}
    \[
        y(x) = y_h(x) + y_p(x)
    \]
\end{mainbox}

\paragraph{Variation der Konstanten}
Die in der homogenen Lösung auftretende Intergrationskonstante \( C \) wird einfach als von \( x \) abhängige Funktion aufgefasst: \( C \to C(x) \).


\subsection{\emph{Lineare} ODEs \emph{\texorpdfstring{\( n \)}{n}-ter Ordnung} mit \emph{konstanten Koeffizienten}}

\subsubsection{Lineare, \emph{homogene} ODEs \texorpdfstring{\( n \)}{n}-ter Ordnung mit konstanten Koeffizienten}

\[
    a_n y^{(n)} + a_{n - 1} y^{(n - 1)} + \dots + a_0 y = 0
\]

Können mit dem Euler-Ansatz gelöst werden:
\[
    y(x) = e^{\lambda x}
\]

Das führt zum charakteristischen Polynom:
\[
    a_n \lambda^n + a_{n - 1} \lambda^{n - 1} + \dots + a_0 = 0
\]

Hat das charakteristische Polynom Nullstellen \( \lambda_1, \, \dots, \, \lambda_k \) mit Vielfachheiten \( m_1, \dots, \, m_k \), so besitzt die Differentialgleichung die linear unabhängigen Lösungen \begin{gather*}
    e^{\lambda_1 x}, \, x \cdot e^{\lambda_1 x}, \, \dots, \, x^{m_1 - 1} \cdot e^{\lambda_1 x} \text{,} \\
    e^{\lambda_2 x}, \, x \cdot e^{\lambda_2 x}, \, \dots, \, x^{m_2 - 1} \cdot e^{\lambda_2 x} \text{,} \\
    \vdots \\
    e^{\lambda_k x}, \, x \cdot e^{\lambda_k x}, \, \dots, \, x^{m_k - 1} \cdot e^{\lambda_k x} \text{,} \\
\end{gather*}
das \emph{Fundamentalsystem}. Die allgemeine Lösung ist eine Linearkombination des Fundamentalsystems. Die Koeffizienten werden dann aus den Anfangsbedingungen bestimmt.

Wenn das charakteristische Polynom komplexe Nullstellen hat, kann man die Basis mit \( e^{ix} = \cos x + i \sin x \) umformen.


\subsubsection{Lineare, \emph{inhomogene} ODEs \texorpdfstring{\( n \)}{n}-ter Ordnung mit konstanten Koeffizienten}

\begin{mainbox}{}
    \[
        y(x) = y_h(x) + y_p(x)
    \]
\end{mainbox}

\paragraph{Direkter Ansatz (educated guess)}: Der Ansatz für \( y_p(x) \) hat dieselbe Form wie der inhomogene Term \( b(x) \). Es können Summen aus Ansätzen kombiniert werden. \( \sin \) kann dabei immer auch mit \( \cos \) ausgetauscht werden.

\begin{table}[H]
    \centering
    \footnotesize
    \begin{tabular}{|c|c|}
    \hline
    \textbf{Inhomogener Term $b(x)$} & \textbf{Ansatz für $y_p(x)$} \\ \hline
    $\sum_{i=0}^m b_ix^i$ & $\sum_{i=0}^m A_i x^i$ \\
    \hline
    $e^{\alpha x} \sum_{i=0}^m b_i x^i$ & $e^{\alpha x} \sum_{i=0}^m A_i x^i$ \\
    \hline
    $\sin(\omega x) \sum_{i=0}^m b_i x^i$ & $\sin(\omega x) \sum_{i=0}^m A_i x^i$ \\
    \hline
    $\sinh(\omega x) \sum_{i=0}^m b_i x^i$ & $\sinh(\omega x) \sum_{i=0}^m A_i x^i$ \\
    \hline
    $e^{\alpha x} \sin(\omega x) \sum_{i=0}^m b_i x^i$ & $e^{\alpha x} \sin(\omega x) \sum_{i=0}^m A_i x^i$ \\ \hline
    \end{tabular}
\end{table}

Wenn ein Teil der für \( y_p(x) \) zu wählenden Funktion bereits in der Lösung des homogenen Problems vorhanden ist, wird der Ansatz solange mit \( x \) multipliziert bis dies nicht mehr der Fall ist.


\section{Differentialrechnung}

\subsection{Stetigkeit}

\begin{mainbox}{Grenzwert einer Funktion}
    Sei \( f: \: \Omega \subseteq \R^n \to \R^m, \, x_0 \in \Omega, \, y \in \R^m \). Die Funktion \( f \) hat Grenzwert \( y \) für \( x \to x_0 \), falls \( \forall \varepsilon > 0, \exists \delta > 0 \), sodass \( \forall x \in \Omega, \, x \neq x_0, \, \| x - x_0 \| < \delta \) gilt \( \| f(x) - y \| < \varepsilon \).
\end{mainbox}

\begin{mainbox}{Konvergenz einer Folge}
    Sei \( \left( x_k \right) \) eine Folge in \( \R^n \), sei \( y = \begin{pmatrix}
        y_1 & \cdots & y_n
    \end{pmatrix} \in \R^n \). Die Folge \( \left( x_k \right) \) konvergiert gegen \( y \) für \( k \to \infty \), falls \( \forall \varepsilon > 0, \, \exists N \geq 1 \), sodass \( \forall k \geq N \) gilt \( \| x_k - y \| < \varepsilon\).
\end{mainbox}

\begin{mainbox}{\( p \)-Norm}
    Die \( p \)-Norm eines reellen oder komplexen Vektors \( x = \begin{pmatrix}
        x_1 & \cdots & x_n
    \end{pmatrix}^\mathsf{T} \in \K^n \) ist für reelles \( 1 \leq p < \infty \) definiert durch \( \| x \|_p = (\sum_{i=1}^n |x_i|^p)^\frac{1}{p} \).
\end{mainbox}

\begin{enumerate}
    \item \( \| x \| > 0 \quad \forall x \neq \mathbf{0} \) (Definitheit)
    \item \( \| t \cdot x \| = |t| \cdot \| x \| \quad t \in \R \) (absolute Homogenität)
    \item \( \| x + y \| \leq \| x \| + \| y \| \) (Subadditivität oder Dreiecksungleichung)
\end{enumerate}

\( \| x \|_2 \) wird als \emph{Euklidische Norm (Distanz)} bezeichnet.

\begin{subbox}{}
    Sei \( (x_k) \) eine Folge in \( \R^n \), \( y = \begin{pmatrix}
        y_1 & \cdots & y_n
    \end{pmatrix} \in \R^n \). Folgende Aussagen sind äquivalent:
    \begin{enumerate}
        \item \( \lim_{k \to \infty} (x_k) = y \)
        \item Für jedes \( 1 \leq i \leq n \) konvergiert die Folge \( (x_k)_i \) gegen \( y_i \in \R \)
        \item \( \lim_{k \to \infty} \| x_k - y \| = 0 \)
    \end{enumerate}
\end{subbox}

Funktionen, die immer stetig sind:
\begin{itemize}
    \item Polynome
    \item Trigonometrische Funktionen (\( \sin \), \( \cos \), \( \tan \), \dots)
    \item Hyperbolische Funktionen (\( \sinh \), \( \cosh \), \( \tanh \))
    \item \( \exp \), \( \log \)
\end{itemize}
\begin{subbox}{}
    Summe, Differenz, Produkt, Quotient (falls der Nenner \( \neq 0 \)) und die Zusammensetzung stetiger Funktionen sind stetig.
\end{subbox}

\begin{subbox}{}
    Differenzierbarkeit \( \implies \) Stetigkeit \( \implies \) Integrierbarkeit
\end{subbox}

Falls \( f: \: \R^2 \to \R \) stetig ist, können wir \( y_0 \) fixieren und die Funktion \( g_{y_0}(x) = f(x, y_0) \) ist stetig. Umgekehrt gilt das nicht!

\begin{mainbox}{Sandwichsatz}
    Falls \( f, g, h \) Funktionen sind mit \( f(x) < g(x) < h(x) \; \forall x \in \R^n \). Sei \( a \in \R^n \). Falls \( \lim_{x \to a} f(x) = \lim_{x \to a} h(x) = L \), dann gilt \( \lim_{x \to a} g(x) = L \).
\end{mainbox}

\paragraph{Polarkoordinaten-Trick}
Sei \( f(x, y) \) eine Funktion. Setze \begin{align*}
    x &= r \cdot \cos \varphi \\
    y &= r \cdot \sin \varphi
\end{align*}
Falls \( \lim_{r \to 0} \) unabhängig von \( \varphi \) ist, d.h. es ist egal, aus welcher Richtung wir uns \( (x_0, y_0) \) nähern, ist \( f(x, y) \) stetig.


\subsection{Partielle Ableitungen}

\begin{mainbox}{}
    Eine partielle Ableitung ist die Ableitung einer Funktion mit mehreren Variablen nach einer dieser Variablen (in Richtung dieser Koordinatenachse). Die Werte der übrigen Argumente werden konstant gehalten.
\end{mainbox}

\begin{mainbox}{Partielle Differenzierbarkeit}
    Eine Funktion \( f: \: \Omega \subseteq \R^n \to \R \) heisst an der Stelle \( a \in \Omega \) nach der Variablen \( x_i \) partiell differenzierbar, falls \begin{multline*}
        \pdv{f}{x_i} a = \\
        \lim_{h \to 0} \frac{f(a_1, \dots, a_i + h, \dots, a_n) - f(a_1, \dots, a_i, \dots, a_n)}{h}
    \end{multline*} existiert.
\end{mainbox}

Nicht stark genug, um die Stetigkeit zu implizieren (wie im eindimensionalen Fall)!

\begin{mainbox}{Gradient}
    \[
        \gradient(f) = \nabla f = \begin{pmatrix}
            \pdv{f}{x_1} \\
            \vdots \\
            \pdv{f}{x_n}
        \end{pmatrix}
    \]
\end{mainbox}

\begin{mainbox}{Hesse-Matrix}
    \[
        \Hess_f = \begin{pmatrix}
            \pdv[order=2]{f}{x_1} & \cdots & \pdv{f}{x_1,x_n} \\
            \vdots & \ddots & \vdots \\
            \pdv{f}{x_n,x_1} & \cdots & \pdv[order=2]{f}{x_n}
        \end{pmatrix}
    \]
\end{mainbox}


\subsection{Der Satz von Schwarz}

\begin{mainbox}{Satz von Schwarz}
    Sei \( f \in C^2(\Omega) \), also \( f \) \( k \)-mal stetig differenzierbar, dann gilt
    \[
        \pdv{f}{x_i, x_j} = \pdv{f}{x_j, x_i}
    \]
\end{mainbox}


\subsection{Vektorwertige Funktionen}

\begin{mainbox}{Partielle Differenzierbarkeit}
    \( f \) heisst an der Stelle \( a \in \Omega \) partiell differenzierbar, falls jede der Komponenten \( f_i \) von \( f \) in \( a \) partiell differenzierbar ist.
\end{mainbox}

\begin{mainbox}{Jacobi-Matrix}
    \[
        J_f = \begin{pmatrix}
            \pdv{f_1}{x_1} & \cdots & \pdv{f_1}{x_n} \\
            \vdots & \ddots & \vdots \\
            \pdv{f_m}{x_1} & \cdots & \pdv{f_m}{x_n}
        \end{pmatrix}
    \]
\end{mainbox}

Für \( f: \: \R^n \to \R \) gilt:
\[ \nabla f(x_0) = (J_f(x_0))^\T \]

\begin{mainbox}{Divergenz}
    Sei \( f: \Omega \subseteq \R^n \to \R^n \) ein Vektorfeld von der Klasse \( C^1 \) mit Komponenten \( f_1, f_2, f_n \).
    \[
        \divergenz(f) = \spur(J_f) = \frac{\partial f_1}{\partial x_1} + \frac{\partial f_2}{\partial x_2} + \dots + \frac{\partial f_n}{\partial x_n}
    \]
\end{mainbox}

\begin{mainbox}{Richtungsableitung}
    Die Richtungsableitung von \( f \) in Richtung \( v = (a, b) \) mit \( \| v \|_2 = 1 \) ist
    \[
        \partial_v f(\mathbf{x}) = \lim_{h \to 0} \frac{f(\mathbf{x} + h \cdot \mathbf{v}) - f(\mathbf{x})}{h}
    \]
\end{mainbox}

Falls \( f \) total differenzierbar ist:
\[
    D_f(x) \cdot \mathbf{v}
\]


\subsection{Das Differential}

\begin{mainbox}{Differential}
    Sei \( X \subseteq \R^n \) offen, \( x_0 \in X \), \( f: \: X \to \R^m \). Die Funktion \( f \) heisst an der Stelle \( x_0 \) (total) differenzierbar, falls es eine lineare Abbildung \( U: \: \R^n \to \R^m \) gibt, so dass \[
        \lim_{x \to x_0} \frac{\| f(x) - f(x_0) - U(x - x_0) \|}{\| x - x_0 \|} = 0 \text{.}
    \]
    Die Abbildung \( U \) heisst \emph{Differential} von \( f \) an der Stelle \( x_0 \).
\end{mainbox}

\begin{subbox}{}
    Sei \( f: \: X \subseteq \R^n \to \R^m \) differenzierbar in \( x_0 \in X \), dann \begin{enumerate}
        \item \( f \) ist stetig in \( x_0 \)
        \item Alle partiellen Ableitungen von \( f \) in \( x_0 \) existieren (und sie sind stetig)
        \item Die Matrix von \( \mathrm{d}f(x_o) \) bzgl. den Standardbasen ist gleich \( J_f(x_0) \).
    \end{enumerate}
\end{subbox}

Wenn also \( J_f(x_0) \) definiert ist, dann ist \( f \) an der Stelle \( x_0 \) differenzierbar und somit auch stetig.

\subsubsection{Bestimmung der Differenzierbarkeit}
\begin{subbox}{}
    Summe, Differenz, Produkt, Quotient (falls der Nenner \( \neq 0 \) und die Zusammensetzung differenzierbarer Funktionen sind differenzierbar.
\end{subbox}
\begin{enumerate}
    \item Es lohnt sich, die Stetigkeit der Funktion zu überprüfen (falls in \( x_0 \) nicht stetig \( \implies \) kann in \( x_0 \) nicht differenzierbar sein)
    \item Auf Intervallen verwendet man das obige Theorem, ansonsten haben wir zwei Möglichkeiten:
    \begin{itemize}
        \item Die Definition der (totalen) Differenzierbarkeit
        \item Die partiellen Ableitungen bestimmen und auf Stetigkeit überprüfen
    \end{itemize}
\end{enumerate}

\begin{subbox}{}
    \begin{itemize}
        \item Total differenzierbare Funktionen sind stetig
        \item Total differenzierbare Funktionen sind partiell differenzierbar
        \item Partiell differenzierbare Funktionen sind nicht notwendigerweise stetig und damit auch nicht notwendigerweise total differenzierbar.
        \item Stetig partiell differenzierbare Funktionen, also Funktionen, deren partielle Ableitungen stetig sind, sind dagegen stetig total differenzierbar.
    \end{itemize}
\end{subbox}


\subsection{Die Kettenregel}

\( f, g: \: \R \to \R \):
\[
    (g \circ f)' (x) = g'(f(x)) \cdot f'(x)
\]

\( f: \: \Omega \subseteq \R^n \to \R^m \) und \( g: \: U \subseteq \R^m \to \R^l \) mit \( U \subseteq f(\Omega) \):
\[
    \odif{(g \circ f)}(x) = \odif{g}(f(x)) \cdot \odif{f}(x)
\]


\subsection{Taylorpolynome}

\begin{mainbox}{Taylorpolynom}
    Sei \( U \subseteq \R^m \) offen, \( f \in C^k(U, \R) \), und sei \( y \in U\).
    \[
        T_k(x) = \sum_{|i| \leq k} \frac{\partial_i f(y) \cdot (x - y)^i}{i!} \text{,}
    \]
    wobei \( i \) ein Multiindex ist:
    \begin{itemize}
        \item \( i \) ist Tupel von \( m \) nicht-negativen ganzen Zahlen.
        \item  \( |i| = i_1 + \dots + i_m \)
        \item \( (x - y)^i = (x_1 - y_1)^{i_1} \dots (x_n - y_n)^{i_n} \)
    \end{itemize}
\end{mainbox}

Das Taylorpolynom 2. Ordnung von \( f \) bei \( \mathbf{x_0} \) ist:
\begin{multline*}
    T_2 f(\mathbf{x}) = \\
    \underbrace{\underbrace{\underbrace{f(\mathbf{x_0})}_{T_0} + \langle \nabla f(\mathbf{x_0}), \mathbf{x} - \mathbf{x_0} \rangle}_{T_1} + \frac{1}{2} (\mathbf{x} - \mathbf{x_0})^\mathsf{T} \cdot H_f(\mathbf{x_0}) \cdot (\mathbf{x} - \mathbf{x_0})}_{T_2}
\end{multline*}

Das Taylorpolynom 3. Ordnung von \( f \) bei \( (x_0, y_0) \) ist:
\begin{multline*}
    T_3 f(x, y) = \\
    \left. \begin{aligned}
        \left. \begin{aligned}
            \left. \begin{aligned}
                \left.
                    f(x_0, y_0)
                \quad \right\} \ T_0 \\
                + \left[ \begin{aligned}
                    &\frac{\partial f}{\partial x} (x_0, y_0)(x - x_0) \\
                    + &\frac{\partial f}{\partial y} (x_0, y_0)(y - y_0)
                \end{aligned} \right]
            \end{aligned} \quad \right\} \ T_1 \\
            + \frac{1}{2} \left[ \begin{aligned}
                &\frac{\partial^2 f}{\partial x^2} (x_0, y_0)(x - x_0)^2 \\
                + &2 \frac{\partial^2 f}{\partial x \partial y} (x_0, y_0)(x - x_0)(y - y_0) \\
                + &\frac{\partial^2 f}{\partial y^2} (x_0, y_0)(y - y_0)^2 \\
            \end{aligned} \right]
        \end{aligned} \quad \right\} \ T_2 \\
        + \frac{1}{6} \left[ \begin{aligned}
            &\frac{\partial^3 f}{\partial x^3} (x_0, y_0)(x - x_0)^3 \\
            + &3 \frac{\partial^3 f}{\partial x^2 \partial y} (x_0, y_0)(x - x_0)^2(y - y_0) \\
            + &3 \frac{\partial^3 f}{\partial x \partial y^2} (x_0, y_0)(x - x_0)(y - y_0)^2 \\
            + &\frac{\partial^3 f}{\partial y^3} (x_0, y_0)(y - y_0)^3
        \end{aligned} \right]
    \end{aligned} \quad \right\} \ T_3
\end{multline*}

\begin{mainbox}{Tayorapproximation}
    Sei \( f \in C^k(U, \R), \, y \in U \). Dann gilt:
    \[
        f(x) = T_k f(x) + o(\| x - y \|^k)
    \]
\end{mainbox}


\subsubsection{Landau-Notation}
\begin{subbox}{}
    Sei \( U \subseteq \R^m, \, h: \: U \to \R, \, y \in U \). Dann:
    \[
        o(h) = \left\{ f: \: U \to \R \: \middle| \: \lim_{\substack{x \to y \\ x \neq y}} \frac{f(x)}{h(x)} = 0 \right\}
    \]
\end{subbox}
\begin{itemize}
    \item \( o(h) \) beschreibt die Menge der Funktionen, die bei \( y \) schneller gegen \( 0 \) gehen als \( h \), wenn \( x \to y \).
    \item Zum Beispiel bedeutet \( f(x) = o(x - y) \), dass \( f(x) \) schneller gegen \( 0 \) geht als \( (x - y) \).
\end{itemize}

\paragraph{Rechenregeln für \texorpdfstring{\( o(h) \)}{o(h)}}
\begin{itemize}
    \item \( \lambda \cdot o(h) + \mu \cdot o(h) \subseteq o(h) \quad \text{für \( \lambda, \mu \in \R \)} \)
    \item \( g \cdot o(h) \subseteq o(g) \cdot o(h) \subseteq o(g \cdot h) \)
    \item \( o(o(h)) \subseteq o(h) \)
    \item \( o(\| x - y \|^d) \subseteq o(\| x - y \|^e) \quad \text{für \( e \leq d\)} \)
\end{itemize}


\subsubsection{Tangentialraum}

\begin{mainbox}{Tangentialraum}
    Sei \( U \subset \R^n \) offen, \( f: \: U \to \R^m \) differenzierbar, \( x_0 \in U, \, A = \mathrm{df}(x_0) = J_f(x_0) \). Der Tangentialraum bei \( x_0 \) des Graphen von \( f \) ist der Graph von \( g(x) = f(x_0) + A(x - x_0) \).
\end{mainbox}

Der Tangentialraum von \( f \) an der Stelle \( x_0 \) ist nichts anderes als das Taylorpolynom 1. Ordnung an der Stelle \( x_0 \).

\begin{subbox}{}
    Sei \( f: \: \R \to \R \) und \( g: \: \R^n \to \R \) in \( C^k, \, k \geq 1 \), und \( g(0) = 0 \). Sei \begin{align*}
        p &= T_k f(y), \quad \text{sodass } f(y) = p(y) + o(y^k) \text{, \quad und} \\
        q &= T_k g(x), \quad \text{sodass } g(x) = q(x) + o(\| x \|^k) \text{.}
    \end{align*}
    Dann:
    \[
        f(g(x)) = p(q(x)) + o(\| x \|^k) \text{.}
    \]
\end{subbox}

Daraus folgt:
\begin{multline*}
    T_k f(g(x)) = \\
    \text{\( p(q(x)) \) nach Entfernen von Monomen von Grad \( > k \)}
\end{multline*}

Die Bedingung \( g(0) = 0 \) ist nötig, weil das Taylorpolynom von \( f \) nur Auskunft über \( f \) nahe bei \( 0 \) (oder allgemeiner dem Entwicklungspunkt) gibt.

Für \( f(y) = e^y, \quad g(x_1, x_2) = \cos(x_1) \cdot \cos(x_2), \quad k = 3 \):
\begin{align*}
    f(y) &= 1 + y + \frac{y^2}{2} + \frac{y^3}{6} + o(y^3) \\
    g(x_1, x_2) &= \left( 1 - \frac{x_1^2}{2} + o(x_1^3) \right) \left( 1 - \frac{x_2^2}{2} + o(x_2^3) \right) \\
        &= 1 - \frac{x_1^2}{2} - \frac{x_2^2}{2} + o(\| x \|^3)
\end{align*}
Problem: \( g(0, 0) = 1 \neq 0 \implies \) Satz lässt sich nicht direkt anwenden. Wenn man einfach trotzdem versucht, ein Taylorpolynom ins andere einzusetzen, stösst man auf Schwierigkeiten:
\[
    p(q(x_1, x_2)) = 1 + \left( 1 - \frac{x_1^2}{2} - \frac{x_2^2}{2} \right) + \frac{\left( 1 - \frac{x_1^2}{2} - \frac{x_2^2}{2} \right)^2}{2} + \dots
\]
Egal, wie hoch man den Grad des Taylorpolynoms von \( f \) wählt: durch Einsetzen von \( g \) erhält man immer weitere konstante Terme.

\paragraph{Trick:}
\begin{gather*}
    f(g(x_1, x_2)) = e^{\cos(x_1) \cos(x_2)} = e \cdot e^{\cos(x_1) \cos(x_2) - 1} \text{,} \\

        \tilde{g}(x_1, x_2) &= \cos(x_1) \cos(x_2) - 1 \quad \text{(neues \( \tilde{g} \) mit \( \tilde{g}(0, 0) = 0 \))} \\

        &= e - e \frac{x_1^2}{2} - e \frac{x_2^2}{2} + o(\| x \|^3).
    \end{aligned}
\end{gather*}





\subsection{Satz von der Umkehrabbildung}

\begin{mainbox}{Satz von der Umkehrabbildung}
    Ist \( \det(J_f(x_0)) \neq 0 \), so ist \( f \) lokal umkehrbar, d.h. es existieren offene Umgebungen \( U \) von \( x_0 \) und \( V \) von \( f(x_0) \), sodass \( f \) eingeschränkt auf \( U \) (\( f: \: U \to V \)) bijektiv ist. Die inverse Abbildung \( f^{-1}: \: V \to U \) ist auf \( V \) von der Klasse \( C^1 \) und für das Differential gilt:
    \[
        \mathrm{d}(f^{-1})(y) = (\mathrm{d}f(x))^{-1} \quad \text{(Inverse von \( J_f \))}
    \]
    wobei \( y = f(x) \).
\end{mainbox}

Siehe \ref{inverse} für das Berechnen der Inverse.

\begin{mainbox}{Diffeomorphismus}
    Eine Abbildung \( f: \: \Omega \to f(\Omega) \) heisst \emph{Diffeomorphismus}, falls \( f \) invertierbar ist und \( f, f^{-1} \) von der Klasse \( C^1 \) sind. \textbf{Anders gesagt:} Ein Diffeomorphismus liegt vor, falls \( f \) umkehrbar ist und sowohl \( f \) als auch \( f^{-1} \) stetig differenzierbar sind.
\end{mainbox}


\subsection{Extremwertaufgaben}

\begin{mainbox}{Kritischer Punkt}
    Ein Punkt \( x_0 \in \R^n \) ist ein kritischer Punkt einer Funktion \( f: \R^n \to \R \), wenn \( \nabla f(x_0) = 0 \).
\end{mainbox}

\begin{subbox}{}
    \begin{align*}
        H_f(x_0) \text{ pos. definit } &\implies x_0 \text{ lokales Min. von } f \\
        H_f(x_0) \text{ neg. definit } &\implies x_0 \text{ lokales Max. von } f \\
        H_f(x_0) \text{ indefinit } &\implies x_0 \text{ Sattelpunkt von } f
    \end{align*}
    \textbf{Achtung: pos. \( \implies \) Min. und neg. \( \implies \) Max.}
\end{subbox}

\begin{subbox}{Eigenwert-Kriterium}
    Alle Eigenwerte von \( H_f(x_0) \) sind \dots
    \begin{itemize}
        \item \dots positiv \( \iff \) \( H_f(x_0) \) ist positiv definit
        \item \dots nicht-negativ \( \iff \) \( H_f(x_0) \) ist positiv semidefinit
        \item \dots negativ \( \iff \) \( H_f(x_0) \) ist negativ definit
        \item \dots nicht-positiv \( \iff \) \( H_f(x_0) \) ist negativ semidefinit
    \end{itemize}
    Hat \( H_f(x_0) \) sowohl positive als auch negative Eigenwerte, ist \( H_f(x_0) \) indefinit.
\end{subbox}

Die Eigenwerte einer quadratischen und symmetrischen Matrix \( A \) bestimmt man, indem man
\[
    \det(A - \lambda \cdot I) = 0
\]
für \( \lambda \) löst. Eine andere Möglichkeit, um die Definitheit einer Matrix zu bestimmen, ist das Sylvester-Kriterium:

\paragraph{Sylvester-Kriterium}
Es seien \( A_1, \dots, A_n \) die Hauptminoren der reellen, symmetrischen \( n \times n \)-Matrix A.
\begin{itemize}
    \item \( \det A_1 > 0, \, \det A_2 > 0, \, \det A_3 > 0, \, \dots \) \( \iff \) \( A \) ist positiv definit
    \item \( \det A_1 < 0, \, \det A_2 < 0, \, \det A_3 < 0, \, \dots \) \( \iff \) \( A \) ist negaitv definit
    \item Sonst ist \( A \) nach diesem Kriterium indefinit
\end{itemize}


\subsubsection{Min-Max-Theorem}

\begin{mainbox}{Min-Max-Theorem}
    Sei \( \bar{X} \subseteq \R^n \) eine kompakte Menge, \( f: \: \bar{X} \to \R \) stetig, dann ist \( f \) beschränkt und hat sein Maximum und Minimum in \( \bar{X} \).
    D.h. \( \exists x^+ \in \bar{X} \) und \( \exists x^- \in \bar{X} \) so dass \[
        f(x^+) = \sup_{x \in \bar{X}} f(x), \quad f(x^-) = \inf_{x \in \bar{X}} f(x)
    \]
\end{mainbox}
Wenn wir Extrema auf einer kompakten Menge suchen, müssen wir auch den Rand auf kritische Punkte untersuchen.

\begin{subbox}{Kompakte Menge}
    \[
        \text{kompakt} \iff \text{abgeschlossen und beschränkt}
    \]
\end{subbox}


\section{Integralrechnung}

\subsection{Wegintegral}

\begin{mainbox}{}
    Es seien \( \vec{v}: \: \Omega \subset \R^n \to \R^n \) ein stetig differenzierbares Vektorfeld und \( \vec{\gamma}: \: [a, b] \to \R^n \) eine differenzierbare Kurve. Der Ausdruck \[
        \int_\gamma \vec{v} \, \mathrm{d}\vec{s} := \int_a^b \vec{v}(\vec{\gamma}(t)) \cdot \vec{\gamma'}(t) \, \mathrm{d}t
    \]
    wird \emph{Wegintegral} (oder Kurvenintegral) von \( \vec{v} \) längs \( \gamma \) genannt.
\end{mainbox}

\begin{itemize}
    \item \textbf{Additivität:} Für \( \gamma = \gamma_1 + \gamma_2 \) gilt
    \[
        \int_{\gamma} \vec{v} \odif{\vec{s}} = \int_{\gamma_1} \vec{v} \odif{\vec{s}} + \int_{\gamma_2} \vec{v} \odif{\vec{s}} \text{.}
    \]
    \item \textbf{Antisymmetrie:}
    \[
        \int_\gamma \vec{v} \odif{\vec{s}} = -\int_{-\gamma} \vec{v} \odif{\vec{s}}
    \]
    \item \textbf{Fundamentalsatz der Differential- und Integralrechnung:}
    \[
        \int_\gamma \vec{v} \odif{\vec{s}} = \int_\gamma \nabla g \odif{\vec{s}} = g(\gamma(b)) - g(\gamma(a))
    \]
    Dies gilt aber nur, falls es ein Potential gibt, bzw. wenn das Vektorfeld konservativ ist.
\end{itemize}
    
Wichtig ist, dass die Kurve \( \vec{\gamma} \) nicht von der Klasse \( C^1 \) sein muss, sondern es genügt, dass \( \vec{\gamma} \) stückweise der Klasse \( C^1 \) angehört. Notiert wird: \( \vec{\gamma} \in C^1_{pw} \).
\begin{subbox}{Stückweise stetige Differenzierbarkeit}
    Eine Kurve \( \vec{\gamma}: \: [a, b] \to \R^n \) heisst \emph{stückweise stetig differenzierbar}, falls \( \vec{\gamma} \) bis auf endlich viele Stellen \( t_0 < t_1 < \dots < t_n \) stetig differenzierbar ist.
\end{subbox}
In diesem Fall gilt einfach \[
    \int_\gamma \vec{v} \, \mathrm{d}s = \sum_{i = 1}^n \int_{\gamma_i} \vec{v} \, \mathrm{d}s \text{.}
\]

\paragraph{Kochrezept für Wegintegrale}

\subparagraph{Gegeben:} Vektorfeld \( \vec{v} \in C^1 \), Kurve \( \gamma \in C^1_{pw} \).

\subparagraph{Gesucht:} Wegintegral \( \int_\gamma \vec{v} \, \mathrm{d}\vec{s} \).

\subparagraph{Schritt 1:} Parametrisiere \( \gamma \), d.h. finde \[
    \vec{\gamma}: \: [a, b] \to \R^n, \, t \mapsto \vec{\gamma}(t) \text{.}
\]

\subparagraph{Schritt 2:} Berechne \( \vec{\gamma}'(t) = \frac{\mathrm{d}}{\mathrm{d}t} \, \vec{\gamma}(t)\) (jede Komponente nach dem Parameter \( t \) ableiten).

\subparagraph{Schritt 3:} Benutze die Formel
\[
    \int_\gamma \vec{v} \, \mathrm{d}\vec{s} = \int_a^b \vec{v}(\vec{\gamma}(t)) \cdot \vec{\gamma}'(t) \, \mathrm{d}t \text{.}
\]

\subsubsection{Konservative Vektorfelder}

\begin{mainbox}{Konservativität}
    Sei \( \Omega \subseteq \R^n \) offen sternförmig, \( f: \: \Omega \to \R^n \) mit \( f \in C^1 \) und \[
        \frac{\partial f_i}{\partial x_j} = \frac{\partial f_j}{\partial x_i} \quad \text{für alle \( i, j = 1 \),}
    \] dann ist \( f \) konservativ.
\end{mainbox}

\begin{subbox}{Rotation (curl)}
    \[
        \rot \vec{F}(x, y, z) = \nabla \times \vec{F} = \begin{pmatrix}
            \pdv{}{x} \\ \pdv{}{y} \\ \pdv{}{z}
        \end{pmatrix} \times \begin{pmatrix}
            F_x \\ F_y \\ F_z
        \end{pmatrix} = \begin{pmatrix}
            \pdv{F_x}{y} - \pdv{F_y}{z} \\
            \pdv{F_x}{z} - \pdv{F_z}{x} \\
            \pdv{F_y}{x} - \pdv{F_x}{y}
        \end{pmatrix}
    \]
\end{subbox}

\[
    \rot(f) = 0 \iff \pdv{f_i}{x_j} = \pdv{f_j}{x_i} \; \forall i, j \iff J_f \text{ ist symmetrisch}
\]

\begin{mainbox}{Potential}
    Es sei das Vektorfeld \( \vec{v}: \: \Omega \subset \R^n \to \R^n \) gegeben. Dann ist \( \vec{v} \) genau dann konservativ, wenn eine differenzierbare Abbildung \( \Phi: \: \Omega \to \R \) existiert mit \[
        \vec{v} = \nabla \Phi \text{.}
    \]
    Ein solches \( \Phi \) heisst \emph{Potential} von \( f \). Falls \( \Omega \) wegzusammenhängend ist, dann ist \( \Phi \) eindeutig bis auf die Integrationskonstante bestimmt.
\end{mainbox}

Wenn ein Vektorfeld konservativ ist, dann gilt
\begin{enumerate}
    \item Das Wegintegral ist wegunabhängig, d.h. es hängt nur vom Anfangs- und Endpunkt von \( \gamma \) und nicht von der exakten Form von \( \gamma \) ab. Man kann also einen komplizierten Weg durch einen einfacheren Weg ersetzen, der denselben Anfangs- und Endpunkt hat (meistens durch ein Liniensegment).
    \item Es gibt ein Potential \( \Phi \) und somit gilt für \( \gamma: \: [a, b] \mapsto \R^n \) der Hauptsatz der Differential- und Integralrechnung
    \[
        \int_\gamma \vec{v} \, \mathrm{d}\vec{s} = \Phi(\gamma(b)) - \Phi(\gamma(a)) \text{.}
    \]
\end{enumerate}

\paragraph{Potential finden:}
\begin{enumerate}
    \item Prüfen ob \( \vec{v} \) konservativ ist.
    \item Es gilt \( \nabla \Phi = \vec{v} \iff \frac{\partial \Phi}{\partial x_1} = v_1, \, \dots, \, \frac{\partial \Phi}{\partial x_n} = v_n \)
    \item Integriere \( v_1 \) nach \( x_1 \), z.B. \( \Phi(x, y, z) = \int v_1 \mathrm{d}x + g(y, z) \), wobei \( g \) von \( x \) unabhängig ist.
    \item Bestimme \( g(y, z) \) indem \( v_2 \) nach \( x_2 \) integriert wird
    \item etc.
    \item Allenfalls Integrationskonstante anpassen, wenn Anfangswerte gegeben sind.
\end{enumerate}


\subsection{Das Riemann-Integral}

\begin{itemize}
    \item \textbf{Kompatibilität:} Für \(n=1,\; U=[a,b]\) gilt
    \[
        \int_U f(x)\,dx
        = \int_{[a,b]} f(x)\,dx
        = \int_a^{b} f(x)\,dx .
    \]
    
    \item \textbf{Linearität:} Für \(f,g:U\to\mathbb{R}\) stetig und \(a,b\in\mathbb{R}\) gilt
    \[
        \int_U \big(a f(x)+b g(x)\big)\,dx
        = a\int_U f(x)\,dx + b\int_U g(x)\,dx .
    \]
    
    \item \textbf{Positivität:} Wenn \(f\le g\), dann gilt
    \[
        \int_U f(x)\,dx \le \int_U g(x)\,dx ,
    \]
    und für \(f\ge 0\) und \(V\subset U\) kompakt gilt
    \[
        \int_V f(x)\,dx \le \int_U f(x)\,dx .
    \]
    
    \item \textbf{Dreiecksungleichung:} Es gilt
    \[
        \left| \int_U f(x)\,dx \right|
        \le \int_U |f(x)|\,dx ,
    \]
    und
    \[
        \int_U |f(x)+g(x)|\,dx
        \le \int_U |f(x)|\,dx + \int_U |g(x)|\,dx .
    \]
    
    \item \textbf{Volumen:} Es gilt
    \[
        \int_U 1 \, dx = \text{Volumen von } U .
    \]
    
    \item \textbf{Additivität bezgl.\ Integrationsbereiche:}
    \[
        U_1,U_2 \subseteq \mathbb{R}^n \text{ kompakt}, \qquad
        f: U_1 \cup U_2 \to \mathbb{R} \text{ stetig}
    \]
    \begin{multline*}
        \int_{U_1 \cup U_2} f(x)\,dx
        = \\ \int_{U_1} f(x)\,dx
        + \int_{U_2} f(x)\,dx
        - \int_{U_1 \cap U_2} f(x)\,dx .
    \end{multline*}
\end{itemize}


\subsubsection{Integration auf Quadern}

\begin{mainbox}{Satz von Fubini}
    Es seien der Quader \( Q = [a_1, b_1] \times [a_2, b_2] \times \dots \times [a_n, b_n] \) und \( f: \: Q \to \R \) mit \( f \in C^0(Q) \) gegeben. Dann gilt \begin{multline*}
        \int_Q f(x) \, \odif{{\mu(x_1, \dots, x_n)}} = \\
        \int_{a_1}^{b1} \int_{a_2}^{b_2} \dots \int_{a_n}^{b_n} f(x_1, x_2, \dots, x_n) \odif{x_n} \cdots \odif{x_2} \odif{x_1} \text{.}
    \end{multline*}
\end{mainbox}


\subsubsection{Integration auf Normalbereichen}

\begin{mainbox}{\( y \)-Normalbereich}
    \( \Omega \) heisst \( y \)-Normalbereich, falls sich \( \Omega \) wie folgt darstellen lässt:
    \[
        \Omega = \left\{ (x, y) \in \R^2 \; \middle| \; a \leq x \leq b, \, f(x) \leq y \leq g(x) \right\} \text{.}
    \] wobei \( f, g \) zwei stetige Funktionen der Variablen \( x \) sind.
\end{mainbox}

Sei
\[
    \Omega = \left\{ (x, y) \in \R^2 \; \middle| \; a \leq x \leq b, \, f(x) \leq y \leq g(x) \right\}
\]
mit stetigen Funktionen \( f, g \) und sei \( F \in C^0(\bar{\Omega})\). Dann gilt
\[
    \int_\Omega F \, \odif{\mu} = \int_a^b \int_{f(x)}^{g(x)} F(x, y) \odif{y} \odif{x}
\]

\begin{mainbox}{\( x\)-Normalbereich}
    \( \Omega \) heisst \( x \)-Normalbereich, falls sich \( \Omega \) wie folgt darstellen lässt:
    \[
        \Omega = \left\{ (x, y) \in \R^2 \; \middle| \; c \leq y \leq d, \, h(y) \leq x \leq k(y) \right\} \text{.}
    \] wobei \( h, k \) zwei stetige Funktionen der Variablen \( y \) sind.
\end{mainbox}

Für
\[
    \Omega = \left\{ (x, y) \in \R^2 \middle| c \leq y \leq d, \, h(y) \leq x \leq k(y) \right\}
\]
gilt analog
\[
    \int_\Omega F \, \odif{\mu} = \int_c^d \int_{h(y)}^{k(y)} F(x, y) \, \odif{x} \, \odif{y}
\]

\begin{mainbox}{Normalbereich (allgemein)}
    \( \Omega \subset \R^n \) heisst Normalbereich, falls sich \( \Omega \) folgendermassen darstellen lässt:
    \begin{multline*}
        \Omega = \{ (x_1, x_2, \dots, x_n) \in \R^n \; | \; a \leq x_1 \leq b, \\
        \, f_1(x_1) \leq x_2 \leq g_1(x_1), \dots, \\
        f_{n - 1}(x_1, x_2, \dots, x_{n - 1}) \leq x_n \leq g_{n - 1}(x_1, x_2, \dots, x_{n - 1}) \}
    \end{multline*} mit stetigen Funktionen \( f_i, g_i \).
\end{mainbox}

Für solche Bereiche gilt die Formel \begin{multline*}
    \int_\Omega f \, \odif{\mu} = \int_a^b \int_{f_1(x_1)}^{g_1(x_1)} \dots \\ \int_{f_{n - 1}(x_1, x_2, \dots, x_{n - 1})}^{g_{n - 1}(x_1, x_2, \dots, x_{n - 1})} f(x_1, x_2, \dots, x_n) \odif{x_n} \dots \odif{x_2} \odif{x_1}
\end{multline*}


\subsubsection{Vernachlässigbare Mengen}
\begin{subbox}{Vernachlässigbare Menge}
    \begin{enumerate}
        \item Für \( 1 \leq m \leq n \) ist eine parametrisierte \( m \)-Menge in \( \R^n \) eine stetige Funktion \[
            f: \: [a_1, b_1] \times \dots \times [a_m, b_m] \to \R^n \text{,}
        \]
        die \( C^1 \) auf \( (a_1, b_1) \times \dots \times (a_m, b_m) \) ist.
        \item \( B \subset \R^n \) heisst \emph{vernachlässigbar}, falls \[
            B \subset \mathrm{Bild}(f_1) \cup \dots \cup \textrm{Bild}(f_k)
        \] für \( m_i \)-Mengen \( f_i \) mit \( m_i < n \).
    \end{enumerate}
\end{subbox}

\begin{subbox}{}
    Ist \( U \subset \R^n \) kompakt und vernachlässigbar, dann
    \[
        \int_U f(x) \, \mathrm{d}x = 0 \quad \text{für alle } f \in C^0(U, \R) \text{.}
    \]
\end{subbox}

\begin{itemize}
    \item \( \R^m \subset \R^n \) ist vernachlässigbar für \( m < n \) .
    \item \( \mathrm{Bild}(\gamma) \) ist vernachlässigbar für einen \( C^1 \)-Weg \( \gamma \) in \( \R^n \), \( n \geq 2 \).
\end{itemize}


\subsubsection{Uneigentliche Integrale}
Wir betrachten zwei Arten von uneigentlichen Integralen in \( R^n \):
\begin{enumerate}
    \item \( f: \: [a, \infty) \times I \to \R \)
    \item \( f: \: \R^2 \to \R \)
\end{enumerate}
\begin{subbox}{Uneigentliches Integral 1. Art}
    Sei \( I \subseteq \R \) ein kompaktes Intervall, \( a \in \R \), und \( f: \: [a, \infty) \times I \to \R \) stetig. Dann
    \[
        \int_{[a, \infty) \times I} f(x, y) \, \odif{x} \, \odif{y} := \lim_{b \to \infty} \int_{[a, b] \times I} f(x, y) \, \odif{x} \, \odif{y}
    \]
\end{subbox}

Ein uneigentliches Integral 1. Art konvergiert oder konvergiert nicht, je nachdem, ob der Grenzwert existiert. Für \( f \geq 0 \) folgt aus dem Satz von Fubini:
\begin{align*}
    \int_{[a, \infty) \times I} f(x, y) \, \odif{x} \, \odif{y} &= \int_a^\infty \left( \int_I f(x, y) \, \odif{y} \right) \, \odif{x} \\
    &= \int_I \left( \int_a^\infty f(x, y) \, \odif{x} \right) \, \odif{y}
\end{align*}
\paragraph{Vorsicht:} \[
    \int_I \left( \int_a^\infty f(x, y) \, \odif{x} \right) \, \odif{y}
\] ergibt für uns nur Sinn, falls \[
    g(y) = \int_a^\infty f(x, y) \, \odif{x}
\] stetig ist und \( g(y) \neq \infty \) für alle \( y \in I \).

\begin{subbox}{Uneigentliches Integral 2. Art}
\begin{align*}
    \int_{\R^2} f(x, y) \, \odif{x} \, \odif{y} &= \int_{-\infty}^\infty \left( \int_{-\infty}^\infty f(x, y) \, \odif{x} \right) \, \odif{y} \\
    &= \int_{-\infty}^\infty \left( \int_{-\infty}^\infty f(x, y) \, \odif{y} \right) \, \odif{x} \\
    &= \lim_{R \to \infty} \int_{B_R(0)} f(x, y) \, \odif{x} \, \odif{y}
\end{align*}
\end{subbox}


\subsubsection{Transformationsformel}

\begin{mainbox}{Transformationsformel}
    Sei \( (x, y) = (g(u, v), h(u, v)) = \Phi(u, v) \) eine Substitution, wobei \( \Phi \) ein \( C^1 \)-Diffeomorphismus ist, d.h. bijektiv und \( \Phi, \Phi^{-1} \in C^1 \), dann gilt
    \[
        \int_\Omega f(x, y) \, \odif{x,y} = \int_{\hat{\Omega}} f(g(u, v), h(u, v)) \left| \det \odif{\Phi} \right| \odif{u, v}
    \]
    Hier ist
    \[
        \odif{\Phi} = J_\Phi
    \]
\end{mainbox}

In \( \R^3 \) ist es analog mit \[ (x, y, z) = (g_1(u, v, w), g_2(u, v, w), g_3(u, v, w)) = \Phi(u, v, w) \] und \( \odif{\Phi} \in \R^{3 \times 3} \).

\subsection{Flächeninhalte mit dem Satz von Green}
\begin{mainbox}{Satz von Green}
    Es seien \( \vec{v} = (v_1, v_2) \) ein stetig differenzierbares Vektorfeld auf einem Gebiet \( \Omega \subset \R^2 \) und \( C \subset \Omega \) ein beschränkter Bereich mit \( C^1_{pw} \)-Rand \( \pdif{C} \).
    Dann gilt:
    \[
        \int_{\partial C} \vec{v} \, \mathrm{d}\vec{s} = \int_C \rot(\vec{v}) \odif{x, y} = \int_C \left( \frac{\partial v_2}{\partial x} - \frac{\partial v_1}{\partial y} \right) \odif{x, y}
    \]
\end{mainbox}

\paragraph{Kochrezept für Flächeninhalte mit dem Satz von Green}
\subparagraph{Gegeben:} \( C \subset \R^2 \) beschränkt mit \( C^1_{pw} \)-Rand \( \partial C \).
\subparagraph{Gesucht:} \( \mu(C) \).
\subparagraph{Schritt 1:} Parametrisiere den Rand von \( C \) mit der Kurve \[
    \gamma: \: [a, b] \to \R^2 \text{.}
\]
\subparagraph{Schritt 2:} Berechne \( \gamma' \) (jede Komponente nach dem Parameter \( t \) ableiten).
\subparagraph{Schritt 3:} Wende den Satz von Green an:
\[
    \int_C 1 \odif{{\mu(x, y)}} = \int_C \vec{v} \odif{x, y} = \int_{\gamma = \pdif{C}} \vec{v} \odif{\vec{s}} = \mu(C) \text{,}
\] mit \( \vec{v} = (0, x) \).

Nun ist \( \partial(f \circ K) / \partial \theta \) der zweite Eintrag in \( J_{f \circ K} \). Dies ist gleich dem Skalarprodukt von \( \nabla_f(0, 1, 0) \) mit der zweiten Spalte von \( J_K(1, \pi / 2, \pi / 2) \). Diese zweite Spalte ist für allgemeine \( r, \theta, \varphi \) gleich
\(
    (-r \sin(\theta) \sin(\varphi), r \cos(\theta) \sin(\varphi), 0)^\mathsf{T} \text{,}
\)
bei den gegebenen Koordinaten also gleich \( (-1, 0, 0) \). Somit ist das Ergebnis \( -1 \).

Finden Sie alle reellwertigen Lösungen der Differentialgleichung
\(
    y''' - 3y'' + y' + 5y = 0 \text{.}
\)
Welche darunter sind nicht-trivial und beschränkt?
\textit{Hinweis:} Erraten Sie eine Nullstelle des charakteristischen Polynoms.
\textbf{Lösung:} Das charakteristische Polynom \( t^3 - 3t^2 + t + 5 \) hat \( t_1 = -1 \) als Nullstelle. Mit Polynomdivision lässt sich das Polynom dann als \( (t + 1)(t^2 - 4t + 5) \) faktorisieren. Die Lösungsformel für quadratische Gleichungen liefert schliesslich \( t_2 = 2 + i \) und \( t_3 = 2 - i \) als weitere Nullstellen. Somit ist die allgemeine Lösung von der Form
\(
    c_1 e^{-x} + c_2 e^{2x} \sin(x) + c_3 e^{2x} \cos(x)
\)
mit \( c_1, c_2, c_3 \in \R \). Keine dieser Lösungen ist nicht-trivial und beschränkt.

Finden Sie eine reellwertige Lösung der Differentialgleichung
\(
    y''' - 3y'' + y' +5y = 25x^2 - 32 + 32 \cos(x) \text{.}
\)
\textbf{Lösung:}
\begin{enumerate*}
    \item Geeigneten Ansatz zur Lösung inhomogener Differentialgleichungen wählen (Polynom und zusätzlich trigonometrische Funktionen).
    Eine solche Partikulärlösung lässt sich mit der Methode der unbestimmten Koeffizienten finden; man findet je eine Partikulärlösung für die Inhomogenitäten \( 25x^2 - 32 \) und \( 32 \cos(x) \).
    \item Für \( x^3 \) ergibt der Ansatz \( y = px^2 + qx + r \), eingesetzt in die ODE, die Gleichung
    \(
        -6p + 2px + q + 5px^2  + 5qx + 5r = 25x^2 - 32 \text{.}
    \)
    Koeffizientenvergleich führt zum linearen Gleichungssystem
    \(
        \begin{aligned}
            -6p + q + 5r &= -32 \\
            2p + 5q &= 0 \\
            5p &= 25
        \end{aligned} \text{,}
    \)
    mit der Lösung \( p = 5, q = -2, r = 0 \), was der Partikulärlösung \( y = 5x^2 - 2x \) entspricht.
    Für die Inhomogenität \( 32 \cos(x) \) braucht man den Ansatz \( P \cos(x) + Q \sin(x) \), der eingesetzt zur Gleichung
    \(
        8P \cos(x) + 8Q \sin(x) = 32 \cos(x) \text{,}
    \)
    und mithin zu \( P = 4, Q = 0 \), also zur Partikulärlösung \( y = 4 \cos(x) \) führt.
    Insgesamt findet man die Partikulärlösung \( y = 5x^2 - 2x + 4 \cos(x) \).
\end{enumerate*}

Finden sie eine Lösung \( f \) der Differetialgleichung in der vorherigen Aufgabe, die \( f(0) = f'(0) = 0 \) erfüllt.
\textbf{Lösung:}
\begin{enumerate*}
    \item Aus der Kombination der ersten zwei Teilaufgaben ergibt sich, dass die allgemeine Lösung der Differentialgleichung von der Form
    \(
        y = c_1 e^{-x} + c_2 e^{2x} \sin(x) + c_3 e^{2x} \cos(x) + 5x^2 - 2x + 4 \cos(x)
    \)
    ist. Man findet \( y(0) = c_1 + c_3 + 4 \), und
    \(
        y' = -c_1 e^{-x} + c_2 e^{2x} (2 \sin(x) + \cos(x)) + c_3 e^{2x} (2 \cos(x) - \sin(x)) + 10x - 2 - \sin(x) / 8 \text{,}
    \)
    also \( y'(0) = -c_1 + c_2 + 2c_3 - 2 \). Folglich ergibt \( y(0) = y'(0) = 0 \) das lineare Gleichungssystem
    \(
        \begin{aligned}
            c_1 + c_3 + 4 &= 0 \\
            -c_1 + c_2 + 2c_3 - 2 &= 0
        \end{aligned}
    \)
    \item Dieses Gleichungssystem hat unendlich viele Lösungen; eine mögliche Lösung ist z.B. \( c_1 = 0, c_2 = 10, c_3 = -4 \), die der Antwort
    \(
        f(x) = 10e^{2x} \sin(x) - 4e^{2x} \cos(x) + 5x^2 - 2x + 4 \cos(x)
    \)
    entspricht.
\end{enumerate*}


\subsection{Aufgabe 4}

Sei \( f: \: (-2, 2) \times (-2, 2) \to \R \) die Funktion
\(
    f(x, y) = e^{\sin(x + y)} + x^2 \text{.}
\)

Begründen Sie, warum \( f \) eine \( C^\infty \)-Funktion ist.
\textbf{Lösung:} Da Polynome, \( \sin \) und \( \exp \) \( C^\infty \) sind, und \( C^\infty \) erhalten bleibt unter Multiplikation und Subtraktion, ist auch \( f \) \( C^\infty \).

Berechnen Sie für alle Punkte im Definitionsbereich den Gradienten und die Hessische von \( f \).
\textbf{Lösung:}
\begin{enumerate*}
    \item \( \nabla f(x, y) = (\cos(x + y) e^{\sin(x + y)} + 2x, \cos(x + y) e^{\sin(x + y)}) \)
    \item \(
        H_f(x, y) = \begin{psmallmatrix} g + 2 & g \\ g & g \end{psmallmatrix}
    \)
    für \( g(x, y) = e^{\sin(x + y)} (\cos(x + y)^2 - \sin(x + y)) \).
\end{enumerate*}

Finden Sie die kritischen Punkte von \( f \).
\textbf{Lösung:} Die kritischen Punkte sind die Nullstellen des Gradienten. Ist der zweite Eintrag des Gradienten gleich \( 0 \), so folgt, dass der erste Eintrag gleich \( 2x \) ist. Ist also auch der erste Eintrag des Gradienten gleich \( 0 \), so folgt \( x = 0\). Dann ist der zweite Eintrag gleich \( \cos(y) e^{sin(y)} \). Dieser Eintrag ist genau dann \( 0 \), wenn einer der Faktoren \( 0 \) ist. Da \( \exp \) keine Nullstellen hat, folgt \( \cos(y) = 0 \). Weil \( y \in (-2, 2) \), gilt dies genau dann wenn \( y = \pm \pi/2 \). Also gibt es zwei kritische Punkte: \( (0, \pi/2) \) und \( (0, -\pi/2) \).

Entscheiden Sie für jeden kritischen Punkt, ob es sich um ein lokales Minimum, ein lokales Maximum oder einen Sattelpunkt handelt.
\textbf{Lösung:} Um diese Entscheidung zu treffen, verwenden wir das Hesse-Kriterium, und entscheiden die Definitheit der \( 2 \times 2\)-Hesse-Matrizen m.H. von Determinante und Spur. Die Hessische hat Determinante \( (g + 2)g - g^2 = 2g \) und Spur \( 2g + 2 \). Man berechnet \( g(0, \pi/2) = -e \) und \( g(0, -\pi/2) = 1/e \). Bei \( (0, \pi/2) \) hat die Hessische also negative Determinante \( = -2e \), somit handelt es sich um einen Sattelpunkt. Bei \( (0, -\pi/2) \) hat die Hessische positive Determinante \( = 2/e \), und positive Spur \( = 2/e + 2 \), somit handelt es sich um ein lokales Minimum.


\subsection{Aufgabe 5}

Sei \( A \subset \R^2 \) das kompakte Gebiet, welches vom Weg \( \gamma: \: [0, 2\pi] \to \R^2 \) umschlossen wird, für
\(
    \gamma(t) = \begin{cases}
        (t, \sin(t) & \text{falls } 0 \leq t < \pi \\
        (2\pi - t, 0) & \text{falls } \pi \leq t \leq 2\pi
    \end{cases} \text{.}
\)
Sei das Vektorfeld \( f: \: \R^2 \to \R^2 \) gegeben durch
\(
    f(x, y) = \left( \sqrt{1 + (\ln(1 + x^2))^2}, -e^x + \sqrt{1 + y^2} \right) \text{.}
\)

Bestimmen Sie Volumen und Schwerpunkt von \( A \).
\textbf{Lösung:}
\begin{enumerate*}
    \item Das Volumen von \( X \) ist gleich \( \int_0^\pi \sin(x) \odif{x} = \left[ -\cos(x) \right]_0^\pi = 2 \).
    \item Entweder: Die Menge \( A \) ist symmetrisch bezüglich der Achse \( x = \pi / 2 \). Deshalb ist die \( x \)-Koordinate des Schwerpunktes von \( A \) aus Symmetriegründen \( x = \pi/2\).
    Oder: Man berechnet die \( x \)-Koordinate des Schwerpunktes mitehilfe des Integrals
    \(
        \frac{1}{\operatorname{vol}(A)} \int_0^\pi \int_0^{\sin(x)} x \odif{y, x}
    \)
    \item Also erhält man die \( y \)-Koordinate des Schwerpunktes als
    \(
        \frac{1}{\operatorname{vol}(A)} \int_0^\pi \int_0^{\sin(x)} y \odif{y, x} = \frac{1}{2} \int_0^\pi \left[ \frac{1}{2} y^2 \right]_0^{\sin(x)} \odif{x} = \frac{1}{2} \int_0^\pi \frac{1}{2} \sin^2(x) \odif{x} = \frac{1}{2} \left[ \frac{x}{4} - \frac{1}{4} \sin(x) \cos(x) \right]_0^\pi = \frac{\pi}{8} \text{.}
    \)
    Der Schwerpunkt ist also bei \( (\pi/2, \pi/8) \).
\end{enumerate*}

Berechnen Sie mit Hilfe des Satzes von Green das Wegintegral
\(
    \int_\gamma f \cdot \odif{\vec{s}}
\)
\textbf{Lösung:}
\begin{enumerate*}
    \item Man beachte, dass \( \gamma \) im Uhrzeigersinn verläuft, d.h. \( A \) liegt rechts von \( \gamma \). Dies ändert das Vorzeichen im Satz von Green.
    \item Also ist
    \(
        \int_\gamma f \cdot \odif{\vec{s}} = \int_A \odv{f_1}{y} - \odv{f_2}{x} \odif{x, y} = \int_A y \odif{x, y} = \int_0^\pi \int_0^{\sin(x)} e^x \odif{y, x} = \int_0^\pi \sin(x) e^x \odif{x} = \left[ \frac{1}{2}(\sin(x) - \cos(x)) e^x \right]_0^\pi = \frac{e^\pi + 1}{2} \text{.}
    \)
\end{enumerate*}

Sei \( g: \: \R^2 \to \R^2 \) das Vektorfeld gegeben durch \( g(x, y) = (0, e^x) \). Berechnen Sie unter Verwendung des Wegintegrals
\(
    \int_\gamma g \cdot \odif{\vec{s}} \text{.}
\)
Zeigen Sie, dass \( f + g \) konservativ ist, und schliessen Sie daraus erneut auf den Wert von \( \int_\gamma f \cdot \odif{\vec{s}} \).
\textbf{Lösung:}
\begin{enumerate*}
    \item Nach der Definition des Wegintegrals ist
    \(
        \int_\gamma g \cdot \odif{\vec{s}} = \int_0^{2\pi} \left\langle \gamma'(t), g(\gamma(t)) \right\rangle \odif{t} = \int_0^\pi \left\langle \begin{psmallmatrix} 1 \\ \cos(t) \end{psmallmatrix}, \begin{psmallmatrix} 0 \\ e^t \end{psmallmatrix} \right\rangle \odif{t} + \int_\pi^{2\pi} \left\langle \begin{psmallmatrix} -1 \\ 0 \end{psmallmatrix}, \begin{psmallmatrix} 0 \\ e^{2\pi - t} \end{psmallmatrix} \right\rangle \odif{t} = \int_0^\pi e^t \cos(t) \odif{t} = \left[ \frac{1}{2}(\sin(t) + \cos(t)) e^t \right]_0^\pi \odif{t} = -\frac{e^\pi + 1}{2}
    \)
    \item Für das Vektorfeld \( f + g \) gilt
    \(
        \odv{(f+g)_1}{y} = \odv{(f + g)_2}{x} = 0
    \)
    und \( A \) ist sternförmig (notwendig für den Punkt). Folglich ist \( f + g \) konservativ.
    \item Also ist
    \(
        \int_\gamma f \cdot \odif{\vec{s}} = \int_\gamma f + g \odif{\vec{s}} - \int_\gamma g \cdot \odif{\vec{s}} = 0 - \left( - \frac{e^\pi + 1}{2} \right) = \frac{e^\pi + 1}{2} \text{.}
    \)
\end{enumerate*}

Sei \( a \in \R \) ein reeller Parameter. Die Funktion
\(
    f(x, y) = \ln \left(a \sin(y) + \sqrt{1 + 2x} \right)
\)
nach \( \R \) sei auf dem Teilgebiet von \( \R^2 \) definiert, auf dem sich der Ausdruck auf der rechten Seite auswerten lässt.

Bestimmen Sie in Abhängigkeit von \( a \) das zweite Taylorpolynom von \( f \) bei \( 0, 0 \).
\textbf{Lösung 1:}
\begin{enumerate*}
    \item Das Taylorpolynom kann als Verknüpfung der Taylorpolynome der Funktionen \( \sin(t) \), \( \sqrt{1 + t} \) und \( \ln(1 + t) \) berechnet werden.
    \item Unter Verwendung der bekannten Taylorreihen für \( \sin(t) \), \( \sqrt{1 + t} \) und \( \ln(1 + t) \) findet man
    \(
        \ln \left(a \sin(y) + \sqrt{1 + 2x} \right) = \ln \left( ay + o(y^2) + 1 + x - x^2/2 + o(x^2) \right) = \ln \left( 1 + (x + ay - x^2/2 + o( \lVert (x, y) \rVert^2 )) \right) = (x + ay - x^2/2) - \frac{1}{2}(x + ay - x^2/2)^2 + o( \lVert (x, y) \rVert^2) = x + ay - x^2 - axy - a^2y^2/2 + o( \lVert (x, y) \rVert^2)) \text{.}
            
    \)
\end{enumerate*}
\textbf{Lösung 2:}
\begin{enumerate*}
    \item Die ersten Ableitungen lauten
    \(
        \partial_x f(x, y) = \frac{1}{\sqrt{1 + 2x} (a \sin(y) + \sqrt{1 + 2x}}
    \)
    und
    \(
        \partial_y f(x, y) = \frac{a \cos(y)}{a \sin(y) + \sqrt{1 + 2x}} \text{.}
    \)
    Die zweiten Ableitungen lauten
    \(
        \partial_x^2 f(x, y) = \frac{-a \sin(y) - 2\sqrt{1 + 2x}}{\sqrt{1 + 2x}^3 (a \sin(y) + \sqrt{1 + 2x})^2}
    \)
    \(
        \partial_y^2 f(x, y) = \frac{a \sin(y) (a \sin(y) + \sqrt{1 + 2x}) + a^2 \cos^2(y)}{(a \sin(y) + \sqrt{1 + 2x})^2}
    \)
    \(
        \partial_x \partial_y f(x, y) = \partial_y \partial_x f(x, y) = \frac{-a \cos(y)}{\sqrt{1 + 2x} (a \sin(y) + \sqrt{1 + 2x})^2}
    \)
\end{enumerate*}

Bestimmen Sie eine Gleichung, welche die Tangentialebene des Graphen von \( f \) bei \( (0, 0) \) beschreibt. Für welche \( a \in \R \) enthält diese Ebene die \( y \)-Achse?
\textbf{Lösung:} Vom ersten Taylorpolynom von \( f \) liest man ab, dass die Tangentialebene durch \( z = x + ay \) gegeben ist. Die \( y \)-Achse ist in dieser Ebene enthalten, wenn alle Punkte auf der \( y \)-Achse, also alle Punkte der Form \( (0, y, 0) \), diese Gleichung erfüllen. Einsetzen ergibt \( 0 = 0 + ay \) für alle \( y \). Dies ist genau dann erfüllt, wenn \( a = 0 \).

Sei \( g: \: (0, 1) \to \R^2 \) die Funktion \( g(t) = ((t^2 - 1)/2, \arcsin(t)) \). Berechnen Sie für alle \( t \in (0, 1) \) das Produkt der Jacobimatrizen
\(
    J_f(g(t)) \cdot J_g(t)
\)
in Abhängigkeit von \( a > 0 \).
\textbf{Lösung 1:} Nach der Kettenregel ist \( J_f(g(t)) \cdot J_g(t) = J_{f \circ g}(t) \). Da \( f \circ g \) eine Funktion \( (0, 1) \to \R \) ist, ist \( J_{f \circ g}(t) \) einfach die \( 1 \times 1 \) Matrix mit Eintrag \( (f \circ g)'(t) \). Man berechnet \( (f \circ g)(t) = \ln((a + 1) t) \), und erhält so als Antwort
\(
    \left( \frac{a + 1}{at + t} \right) = \left( 1/t \right) \text{.}
\)
\textbf{Lösung 2:} Es ist
\(
    J_f(x, y) = \begin{psmallmatrix} \frac{1}{\sqrt{1 + 2x} (a \sin(y) + \sqrt{1 + 2x})} & \frac{a \cos(y)}{a \sin(y) + \sqrt{1 + 2x}} \end{psmallmatrix} \text{.}
\)
Dementsprechend erhalten wir
\(
    J_f(g(t)) = \begin{psmallmatrix} \frac{1}{t(at + t)} & \frac{a \cos(\arcsin(t)}{at + t} \end{psmallmatrix} = \begin{psmallmatrix} \frac{1}{t(at + t)} & \frac{a \sqrt{1 - t^2})}{at + t} \end{psmallmatrix}
\)
wobei wir verwendet haben, dass \( \cos(\arcsin(t)) = \sqrt{1 - t^2} \).
Weiterhin ist
\(
    J_g(t) = \begin{psmallmatrix} t \\ \frac{1}{\sqrt{1 - t^2}} \end{psmallmatrix}
\)
Also erhalten wir
\(
    J_f(g(t)) \cdot J_g(t) = \begin{psmallmatrix} \frac{1}{t(at + t)} & \frac{a \sqrt{1 - t^2}}{at + t} \end{psmallmatrix} \begin{psmallmatrix} t \\ \frac{1}{\sqrt{1 - t^2}} \end{psmallmatrix} = \left( \frac{a + 1}{at + t} \right) = \left( 1/t \right) \text{.}

\end{gather*}


\subsection{Wichtige Taylorreihen} \label{taylorreihen}

\begin{align*}
    e^x &= \sum_{n=0}^\infty \frac{x^n}{n!} &\quad \text{für alle } x \in \R \\
    \ln(x) &= \sum_{n=1}^\infty \frac{(-1)^{n+1}}{n} (x - 1)^n &\quad \text{für } 0 < x \leq 2 \\
    \sin(x) &= \sum_{n=0}^\infty (-1)^n \frac{x^{2n+1}}{(2n + 1)!} &\quad \text{für alle } x \\
    \cos(x) &= \sum_{n=0}^\infty (-1)^n \frac{x^{2n}}{(2n)!} &\quad \text{für alle } x \\
    \tan(x) &= \sum_{n=1}^\infty \frac{B_{2n} (-4)^n (1-4^n)}{(2n)!} x^{2n-1} &\quad \text{für } \lvert x \rvert < \frac{\pi}{2} \\
    \arcsin(x) &= \sum_{n=0}^\infty \frac{(2n)!}{4^n (n!)^2 (2n +1)} x^{2n + 1} &\quad \text{für } \lvert x \rvert < 1 \\
    \arccos(x)& = \frac{\pi}{2} - \arcsin(x) &\quad \text{für } \lvert x \rvert \leq 1 \\
    \arctan(x) &=\sum_{n=0}^\infty (-1)^n \frac{1}{2n + 1} x^{2n + 1} &\quad \text{für } \lvert x \rvert \leq 1
\end{align*}
Hierbei ist \(  B_{2n} \) die \( 2n \)-te Bernoulli-Zahl.
